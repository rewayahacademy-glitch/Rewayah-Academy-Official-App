const { chromium } = require('playwright');
(async () => {
  const out = process.argv[2];
  const b = await chromium.launch();
  for (const [nom, q] of [['praxis-profil-violet', ''], ['praxis-profil-clair', '&fond=clair']]) {
    const p = await b.newPage({ viewport: { width: 1000, height: 1000 }, deviceScaleFactor: 2 });
    await p.goto('http://localhost:8766/index.html?planche=avatar' + q);
    await p.waitForTimeout(9000);
    await p.screenshot({ path: out + '/' + nom + '.png' });
    await p.close();
  }
  await b.close();
})();
