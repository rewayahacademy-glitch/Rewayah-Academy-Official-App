import 'package:flutter/foundation.dart';

/// Langue de la maquette : en arabe, toute la mise en page passe de droite à gauche.
final ValueNotifier<bool> arabe = ValueNotifier<bool>(false);

String tx(String fr, String ar) => arabe.value ? ar : fr;

/// Les gestes de Praxis.
enum Geste {
  aucun, salut, joie, montre, etire, balaie, range, souffle, presque, suivre, cherche, pied, pause, reflexion, porte, cle,
  desole, hoche, surPlace, allume, pointe, gene, un, envoie, coeur, applaudit, baille, genoux, chapeau, non, compte, respire,
}

/// Les visages de Praxis.
enum Expr { neutre, sourire, joie, etonne, reflexion, desole, attente, paisible, gene, concentre, baille }

class Replique {
  const Replique(this.fr, this.ar, this.geste, this.expr, {this.tard = false});
  final String fr, ar;
  final Geste geste;
  final Expr expr;

  /// Une réplique « de fin » ne vient qu'après 55 %.
  final bool tard;
  String get texte => arabe.value ? ar : fr;
}
