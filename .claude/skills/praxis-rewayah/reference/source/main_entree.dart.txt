Sc premier(EcranId e) => switch (e) { EcranId.connexion => Sc.arrivee, EcranId.chargement => Sc.chargeNormal, EcranId.miseAJour => Sc.majNormale };

/// La ligne des scénarios se fait glisser à la souris comme au doigt.
class _Glisse extends MaterialScrollBehavior {
  const _Glisse();
  @override
  Set<ui.PointerDeviceKind> get dragDevices => <ui.PointerDeviceKind>{ui.PointerDeviceKind.touch, ui.PointerDeviceKind.mouse, ui.PointerDeviceKind.trackpad, ui.PointerDeviceKind.stylus};

/// Le cadre : un téléphone (bord sombre, coin continu, îlot) ou une fenêtre d'ordinateur.
class Cadre extends StatelessWidget {
  const Cadre({super.key, required this.telephone, required this.taille, required this.child});
  final bool telephone;
  final Size taille;
  final Widget child;
  @override
  Widget build(BuildContext context) {
    final double marge = telephone ? 12 : 10, r = telephone ? 58 : 30;
    return Container(
      padding: EdgeInsets.all(marge),
      decoration: ShapeDecoration(color: T.s ? const Color(0xFF0B0A12) : T.nuit, shape: coin(r)),
      child: ClipRSuperellipse(
        borderRadius: BorderRadius.all(Radius.circular(r - marge)),
        child: SizedBox.fromSize(size: taille, child: child),
      ),
    );
  }
}

enum Forme { principale, envoye, lienMort }

// ───────────────────────── L'écran ─────────────────────────
class EcranEntree extends StatefulWidget {
  const EcranEntree({super.key, required this.ecran, required this.depart, required this.taille, required this.large, required this.reduit, required this.arabe, required this.enCours});
  final EcranId ecran;
  final Sc? depart;
  final Size taille;
  final bool large, reduit, arabe;
  final ValueNotifier<Sc?> enCours;
  @override
  State<EcranEntree> createState() => EcranEntreeState();
}

class EcranEntreeState extends State<EcranEntree> with SingleTickerProviderStateMixin<EcranEntree> {
  final Theatre th = Theatre();
  late final Ticker _t = createTicker(_tic);
  Duration _avant = Duration.zero;
  final ValueNotifier<double> hautScene = ValueNotifier<double>(0);
  final ValueNotifier<int> pct = ValueNotifier<int>(0);
  final GlobalKey cleScene = GlobalKey(), cleTexte = GlobalKey();
  Sc sc = Sc.arrivee;
  Bas piedEtat = Bas.cours;
  String saisi = '';
  bool appOuverte = false, relance = false;
  int compte = 30, secousse = 0, _derniereReaction = -1;
  Timer? _minuterie;
  final math.Random _h = math.Random();

  bool _premier = true;
  bool get plein => widget.ecran != EcranId.connexion;
  bool get rtl => widget.arabe;
  Corps get px => th.px;

  @override
  void initState() {
    super.initState();
    th.reduit = widget.reduit;
    sc = widget.depart ?? premier(widget.ecran);
    _geo();
    _t.start();
    WidgetsBinding.instance.addPostFrameCallback((_) => _demarrer());
  }

  @override
  void didUpdateWidget(EcranEntree oldWidget) {
    super.didUpdateWidget(oldWidget);
    final EcranEntree o = oldWidget;
    if (o.ecran != widget.ecran || o.taille != widget.taille || o.arabe != widget.arabe || o.reduit != widget.reduit) {
      th.reduit = widget.reduit;
      th.px.reduit = widget.reduit;
      if (o.ecran != widget.ecran) sc = premier(widget.ecran);
      _geo();
      _demarrer();
    }
  }

  void _demarrer() {
    final Map<String, String> q = Uri.base.queryParameters;
    final Sc s = _premier && widget.depart != null ? widget.depart! : (sc.ecran == widget.ecran ? sc : premier(widget.ecran));
    jouer(s, neuf: true);
    if (_premier && q.containsKey('touche')) {
      final int v = int.tryParse(q['touche'] ?? '') ?? 0;
      Future<void>.delayed(Duration(milliseconds: int.tryParse(q['toucheA'] ?? '') ?? 6000), () {
        if (mounted) toucher(variante: v);
      });
    }
    // Pour vérifier les transitions : `?puis=googleFermee&puisA=9000` joue un second scénario après un délai.
    if (_premier && q.containsKey('puis')) {
      Future<void>.delayed(Duration(milliseconds: int.tryParse(q['puisA'] ?? '') ?? 9000), () {
        if (!mounted) return;
        for (final Sc x in Sc.values) {
          if (x.name == q['puis'] && x.ecran == widget.ecran) jouer(x);
        }
      });
    }
    _premier = false;
  }

  @override
  void dispose() {
    _minuterie?.cancel();
    th.gen++;
    _t.dispose();
    super.dispose();
  }

  Rect get carteLarge {
    final Size s = widget.taille;
    const double w = 440, h = 560;
    final double x = rtl ? 72 : s.width - w - 72;
    return Rect.fromLTWH(x, (s.height - h) / 2, w, h);
  }

  double get hautClavier => widget.taille.height * 0.30;

  Forme get forme => switch (sc) {
        Sc.envoye || Sc.attenteLongue => Forme.envoye,
        Sc.expire || Sc.dejaUtilise => Forme.lienMort,
        _ => Forme.principale,
      };

  (String, Ton)? get message => switch (sc) {
        Sc.adresse => (Lib.erreurAdresse, Ton.erreur),
        Sc.reseauLent => (Lib.reseauLent, Ton.info),
        Sc.trop => (Lib.erreurTropTot, Ton.info),
        Sc.horsLigne => (Lib.horsLigne, Ton.erreur),
        Sc.retourLigne => (Lib.retabli, Ton.bon),
        Sc.serveur => (Lib.erreurServeur, Ton.erreur),
        Sc.googleFermee => (Lib.googleFermee, Ton.info),
        Sc.googleRefuse => (Lib.googleRefuse, Ton.erreur),
        _ => null,
      };

  int _lignes(String t, double largeur, double corps) => (t.length * corps * (rtl ? 0.47 : 0.5) / largeur).ceil().clamp(1, 6);

  /// La hauteur dont le formulaire a besoin sur téléphone.
  double get besoin {
    final double lc = widget.taille.width - 48;
    if (th.clavier.cible > 0.5) return 244;
    switch (forme) {
      case Forme.envoye:
        return 28 + 36 + 12 + _lignes(Lib.envoyeDetail, lc, 16) * 24 + (sc == Sc.attenteLongue ? 14 + 22 + _lignes(Lib.encartAttente, lc - 28, 14.5) * 22 : 0) + 22 + 56 + 10 + 30 + 18;
      case Forme.lienMort:
        return 28 + 36 + 8 + _lignes(sc == Sc.expire ? Lib.expireDetail : Lib.dejaDetail, lc, 16) * 24 + 18 + 84 + 16 + 56 + 20;
      case Forme.principale:
        double b = 28 + 36 + 18 + 84 + 16 + 56 + 12 + 56 + 20;
        final (String, Ton)? m = message;
        if (m != null) b += 12 + 22 + _lignes(m.$1, lc - 28, 14.5) * 22;
        if (sc == Sc.session) b += 24 + 22 + _lignes(Lib.sessionPhrase, lc - 28, 14.5) * 22 + 16;
        if (m == null && sc != Sc.session) b += 38;
        return b;
    }
  }

  void _geo() {
    final Size s = widget.taille;
    double hs = s.height;
    CadreScene c;
    if (widget.large) {
      c = plein ? CadreScene.largePlein : CadreScene.large;
    } else if (plein) {
      c = CadreScene.telPlein;
    } else {
      c = CadreScene.tel;
      // La scène garde 38 % de la hauteur au moins, 56 % au plus.
      hs = (s.height - hautClavier * th.clavier.x - _besoinLisse + 26).clamp(s.height * 0.38, s.height * 0.56);
    }
    th.geo = Geo(s, c, rtl, hs, carteLarge, cam: th.cam.x);
    if ((hautScene.value - hs).abs() > 0.3) hautScene.value = hs;
  }

  double _besoinLisse = 362;
  void _tic(Duration d) {
    final double dt = (d - _avant).inMicroseconds / 1e6;
    _avant = d;
    if (dt <= 0) return;
    // Le formulaire change de hauteur en douceur : la scène suit.
    final double b = besoin;
    _besoinLisse += (b - _besoinLisse) * (1 - math.exp(-dt * 9));
    if ((b - _besoinLisse).abs() < 0.2) _besoinLisse = b;
    _geo();
    if (sc == Sc.saisie && !plein) _suivreTexte();
    th.avancer(dt / _ralenti);
  }

  /// Pour l'examen image par image : `?ralenti=8` joue la scène huit fois plus lentement.
  final double _ralenti = math.max(0.1, double.tryParse(Uri.base.queryParameters['ralenti'] ?? '') ?? 1);

  /// Ses yeux suivent la fin du texte tapé.
  void _suivreTexte() {
    final RenderBox? b = cleTexte.currentContext?.findRenderObject() as RenderBox?;
    final RenderBox? scn = cleScene.currentContext?.findRenderObject() as RenderBox?;
    if (b == null || scn == null || !b.attached) return;
    final Offset caret = scn.globalToLocal(b.localToGlobal(Offset(b.size.width, b.size.height / 2)));
    final Offset d = caret - th.centrePraxis;
    px.regarder(d.dx / 130, d.dy / 110);
  }

  // ───── Mise en scène ─────
  Future<void> _scene(Future<void> Function() f) async {
    try {
      await f();
    } on Annule {
      // Une autre scène a commencé.
    }
  }

  /// D'où qu'il soit, il se relève, se remet de face et revient au premier plan, sans saut.
  Future<void> _revenir(int g, {double travers = 0}) async {
    th.taire();
    px.faire(Geste.aucun);
    px.regarder(0, 0);
    th.eclat.cible = 1;
    if (px.assis.cible > 0.5) {
      px.assis.cible = 0;
      px.ecrase.v -= 1.2;
      await th.attendre(380, g);
    }
    if (px.tourne.cible != 0) {
      px.tourne.cible = 0;
      px.pietine = 2;
    }
    if ((px.q - 1).abs() > 0.01 || (px.lat.x - travers).abs() > 0.03) {
      px.anticipe();
      await th.attendre(130, g);
      px.allerA(1, travers: travers, vitesse: 0.32);
      await th.jusqua(() => px.arrive, g);
    }
  }

  /// Le bord de la route où il s'écarte : côté opposé au formulaire en version large.
  double get bord => widget.large ? (rtl ? 0.62 : -0.62) : (rtl ? -0.4 : 0.4);

  void jouer(Sc e, {bool neuf = false}) {
    _minuterie?.cancel();
    const Set<Sc> entrees = <Sc>{Sc.arrivee, Sc.habitue, Sc.session, Sc.chargeNormal, Sc.chargeLent, Sc.chargeEchec, Sc.majDispo, Sc.maintenance};
    final bool depuisPorte = neuf || appOuverte || !th.present || entrees.contains(e) || sc == Sc.maintenance || sc.ecran != e.ecran;
    if (depuisPorte) {
      th.initial();
    } else {
      th.gen++;
    }
    final int g = th.gen;
    final Sc avant = sc;
    th.clavier.cible = e == Sc.saisie ? 1 : 0;
    widget.enCours.value = e;
    setState(() {
      sc = e;
      appOuverte = false;
      relance = false;
      piedEtat = Bas.cours;
      saisi = switch (e) {
        Sc.arrivee || Sc.saisie => '',
        Sc.adresse => 'amina.bexemple.com',
        _ => 'amina.b@exemple.com',
      };
      if (e == Sc.adresse) secousse++;
    });
    _scene(() => switch (e.ecran) {
          EcranId.connexion => _connexion(e, g, depuisPorte, avant),
          EcranId.chargement => _chargement(e, g),
          EcranId.miseAJour => _miseAJour(e, g, depuisPorte),
        });
  }

  Future<void> _connexion(Sc e, int g, bool depuisPorte, Sc avant) async {
    final bool court = e == Sc.habitue || e == Sc.session;
    if (depuisPorte) {
      await th.arriver(g, vite: court || e != Sc.arrivee);
    } else {
      await _revenir(g);
    }
    Future<void> att(int ms) => th.attendre(ms, g);
    switch (e) {
      case Sc.arrivee:
        px.faire(Geste.salut);
        px.poser(Expr.sourire);
        await att(250);
        await th.parler(Dit.bonjour, g, tenir: 1.2);
        px.faire(Geste.pointe);
        px.regarder(0, 0.9);
        await th.parler(Dit.adresseEtPorte, g, tenir: 2.0);
        px.faire(Geste.aucun);
        px.regarder(0, 0);
      case Sc.habitue:
        px.faire(Geste.hoche);
        px.poser(Expr.joie);
        await th.parler(Dit.habitue, g, tenir: 1.1);
        px.faire(Geste.pointe);
        px.poser(Expr.sourire);
        px.regarder(0, 0.9);
        await th.parler(Dit.habitue2, g, tenir: 1.4);
        px.faire(Geste.aucun);
        px.regarder(0, 0);
      case Sc.session:
        px.poser(Expr.desole);
        px.faire(Geste.desole);
        await att(250);
        await th.parler(Dit.session, g);
        px.faire(Geste.pointe);
        px.poser(Expr.sourire);
        px.regarder(0, 0.9);
        await att(1600);
        px.faire(Geste.aucun);
        px.regarder(0, 0);
      case Sc.saisie:
        px.poser(Expr.neutre);

  /// Chargement : il sort, dit ce qu'il fait, repart quand c'est prêt.
  Future<void> _chargement(Sc e, int g) async {
    final double debut = th.temps;
    Future<void> att(int ms) => th.attendre(ms, g);
    await th.arriver(g, vite: true);
    px.faire(Geste.salut);
    px.poser(Expr.sourire);
    await th.parler(Dit.ouverture, g, tenir: 1.0);
    px.faire(Geste.aucun);
    if (e == Sc.chargeEchec) {
      await th.jusqua(() => th.temps - debut > 5.2, g);
      th.eclat.cible = 0.3;
      await att(400);
      px.poser(Expr.desole);
      px.faire(Geste.desole);
      setState(() => piedEtat = Bas.echecCharge);
      await th.parler(Dit.chargeEchec, g);
      px.faire(Geste.pointe);
      px.regarder(0, 0.9);
      await att(1500);
      px.faire(Geste.aucun);
      px.regarder(0, 0);
      return;
    }
    if (e == Sc.chargeLent) {
      // Après huit secondes, il le dit : d'autres phrases, d'autres gestes.
      await th.jusqua(() => th.temps - debut > 8, g);
      setState(() => piedEtat = Bas.lent);
      int i = 0;
      while (th.temps - debut < 17.5) {
        final Replique r = phrasesLenteur[i % phrasesLenteur.length];
        i++;
        px.poser(r.expr);
        px.faire(r.geste);
        if (r.geste == Geste.montre) px.regarder(-0.5, 0.7);
        await th.parler(r.texte, g, tenir: 1.5);
        px.faire(Geste.pied);
        px.regarder(0, 0);
        await att(700);
      }
    } else {
      await th.jusqua(() => th.temps - debut > 4.4, g);
    }
    setState(() => piedEtat = Bas.cours);
    px.poser(Expr.joie);
    px.faire(Geste.suivre);
    await th.parler(Dit.pret, g, tenir: 0.8);
    await th.repartir(g, ouvert: () => setState(() => appOuverte = true));
  }

  /// Mise à jour : les dalles s'allument de Praxis vers la porte, il parle pour faire patienter.
  Future<void> _miseAJour(Sc e, int g, bool depuisPorte) async {
    Future<void> att(int ms) => th.attendre(ms, g);
    void avancement(double v) {
      th.progres = v;
      pct.value = (v * 100).floor();
    }

    if (e == Sc.maintenance) {
      // La porte est en veilleuse ; il balaie la route, à mi-chemin.
      setState(() => piedEtat = Bas.maintenance);
      th.eclat.cible = 0.5;
      await att(300);
      th.eclat.cible = 0.5;
      px.poser(Expr.paisible);
      px.allerA(0.93, travers: 0, vitesse: 0.42);
      await th.jusqua(() => px.arrive, g);
      px.faire(Geste.balaie);
      await att(1600);
      px.poser(Expr.sourire);
      await th.parler(Dit.maintenance, g);
      px.poser(Expr.paisible);
      return;
    }
    if (e == Sc.majDispo) {
      setState(() => piedEtat = Bas.dispo);
      await th.arriver(g, vite: true);
      px.poser(Expr.joie);
      px.faire(Geste.porte);
      px.regarder(rtl ? 0.4 : -0.4, -0.8);
      await th.parler(Dit.majDispo, g);
      px.faire(Geste.pointe);
      px.poser(Expr.sourire);
      px.regarder(0, 0.9);
      await att(1600);
      px.faire(Geste.aucun);
      px.regarder(0, 0);
      return;
    }
    final double secondes = switch (e) { Sc.majLente => 60, Sc.majVite => 10, Sc.majPresque => 40, Sc.majEchec => 14, _ => 30 };
    double v = switch (e) { Sc.majPresque => 0.86, Sc.majFinie => 0.97, Sc.majEchec => th.progres > 0 ? th.progres : 0.34, _ => 0 };
    final double plafond = e == Sc.majEchec ? 0.62 : 1;
    avancement(v);
    if (depuisPorte) {
      await th.arriver(g, vite: true);
    } else {
      await _revenir(g);
    }
    if (e == Sc.majNormale || e == Sc.majLente || e == Sc.majVite) {
      px.poser(Expr.joie);
      px.faire(Geste.allume);
      await att(450);
      th.etincelle();
      await th.parler(Dit.majDepart, g, tenir: 0.7);
    }
    // La progression n'est pas une droite : elle avance par à-coups, comme une vraie.
    DateTime dernierChiffre = DateTime.now();
    _minuterie = Timer.periodic(const Duration(milliseconds: 120), (Timer t) {
      if (!mounted || g != th.gen) return t.cancel();
      v = math.min(plafond, v + 0.12 / secondes * (0.2 + _h.nextDouble() * 1.6));
      th.progres = v;
      // Le rouleau a besoin d'un demi-temps pour tourner : le chiffre ne change qu'après.
      final DateTime m = DateTime.now();
      if (v >= plafond || m.difference(dernierChiffre).inMilliseconds > 1100) {
        dernierChiffre = m;
        pct.value = (v * 100).floor();
      }
      if (v >= plafond) t.cancel();
    });
    int dernier = -1;
    final List<int> vues = <int>[];
    while (th.progres < plafond - 0.0001) {
      final List<int> possibles = <int>[
        for (int i = 0; i < phrasesAttente.length; i++)
          if (i != dernier && (phrasesAttente[i].tard == (th.progres >= 0.72) || th.progres >= 0.55 && th.progres < 0.72) && !vues.contains(i)) i
      ];
      if (possibles.isEmpty) {
        vues.clear();
        await att(60);
        continue;
      }
      final int i = possibles[_h.nextInt(possibles.length)];
      dernier = i;
      vues.add(i);
      final Replique r = phrasesAttente[i];
      px.poser(r.expr);
      px.faire(r.geste);
      if (r.geste == Geste.porte) px.regarder(rtl ? 0.5 : -0.5, -0.9);
      if (r.geste == Geste.montre) px.regarder(-0.5, 0.7);
      if (r.geste == Geste.compte || r.geste == Geste.balaie) px.regarder(0.2, 0.9);
      if (r.geste == Geste.allume) {
        unawaited(_scene(() async {
          await att(450);
          th.etincelle();
        }));
      }
      if (r.geste == Geste.souffle) th.eclat.cible = 1.22;
      await th.parler(r.texte, g, tenir: 1.9);
      th.eclat.cible = 1;
      px.faire(Geste.aucun);
      px.regarder(0, 0);
      await att(380);
    }
    if (e == Sc.majEchec) {
      // Les dalles s'éteignent en remontant vers lui ; la porte faiblit.
      setState(() => piedEtat = Bas.echecMaj);
      px.sursaut();
      px.poser(Expr.etonne);
      th.eclat.cible = 0.45;
      final double v0 = th.progres;
      for (int i = 1; i <= 24; i++) {
        th.progres = v0 * (1 - i / 24);
        await att(45);
      }
      px.poser(Expr.desole);
      px.faire(Geste.desole);
      await th.parler(Dit.majEchec, g);
      px.faire(Geste.pointe);
      px.regarder(0, 0.9);
      await att(1500);
      px.faire(Geste.aucun);
      px.regarder(0, 0);
      return;
    }
    setState(() => piedEtat = Bas.finie);
    pct.value = 100;
    px.poser(Expr.joie);
    px.faire(Geste.joie);
    await att(700);
    px.faire(Geste.suivre);
    await th.parler(Dit.pret, g, tenir: 0.9);
    await th.repartir(g, ouvert: () => setState(() => appOuverte = true));
  }

  /// On le touche : il réagit, jamais deux fois de suite pareil.
  void toucher({int? variante}) {
    if (!th.present || appOuverte) return;
    int i = variante ?? _h.nextInt(reactionsToucher.length);
    if (variante == null && i == _derniereReaction) i = (i + 1 + _h.nextInt(reactionsToucher.length - 1)) % reactionsToucher.length;
    i %= reactionsToucher.length;
    _derniereReaction = i;
    final Replique r = reactionsToucher[i];
    final int g = th.gen;
    px.ecrase.v += 1.4;
    px.pendV += 6;
    px.poser(r.expr);
    px.faire(r.geste);
    if (r.geste == Geste.aucun) px.sursaut();
    if (r.geste == Geste.non && !th.reduit && px.tourne.cible == 0) {
      // Un tour complet sur lui-même, en sautant.
      px.sauter(300);
      px.tourne.x = 2;
      px.pietine = 4;
    }
    if (r.texte.isNotEmpty) th.dire(r.texte, tenir: 1.0);
    _scene(() async {
      await th.attendre(1900, g);
      // Il reprend sa contenance, sauf si la scène en cours lui a déjà donné autre chose à faire.
      if (px.expr == r.expr) px.poser(Expr.sourire);
      if (px.geste == r.geste) px.faire(Geste.aucun);
    });
  }

  void _pointeur(PointerUpEvent e) {
    final RenderBox? b = cleScene.currentContext?.findRenderObject() as RenderBox?;
    if (b == null) return;
    final Offset p = b.globalToLocal(e.position);
    if ((p - th.centrePraxis).distance < th.largeurPraxis * 0.5) toucher();
  }

final List<(Path, Color)> _google = <(Path, Color)>[
  for (final (String d, Color c) in _tracesGoogle) (_cheminSvg(d), c),

Path _cheminSvg(String d) {
  final Path p = Path();
  for (final CommandeTrace c in lireTraceSvg(d)) {
    final List<double> k = c.coordonnees;
    switch (c.nature) {
      case NatureCommande.deplacer:
        p.moveTo(k[0], k[1]);
      case NatureCommande.ligne:
        p.lineTo(k[0], k[1]);
      case NatureCommande.cubique:
        p.cubicTo(k[0], k[1], k[2], k[3], k[4], k[5]);
      case NatureCommande.quadratique:
        p.quadraticBezierTo(k[0], k[1], k[2], k[3]);
      case NatureCommande.fermer:
        p.close();
    }
  }
  return p;
}

// ───────────────────────── Planche de vérification ─────────────────────────
/// `?planche=1` : le Praxis d'origine (vivant.dart) à gauche, celui de la maquette à droite,
/// dans chaque expression ; puis la toque sous tous les angles.
class PlancheApp extends StatelessWidget {
  const PlancheApp({super.key});
  @override
  Widget build(BuildContext context) {
    final ValueNotifier<Offset> doigt = ValueNotifier<Offset>(Offset.zero);
    Corps pose(Expr e, {double tour = 0, Geste geste = Geste.aucun, bool assis = false}) {
      final Corps c = Corps()
        ..placer(1)
        ..poser(e)
        ..faire(geste);
      c.tourne.x = c.tourne.cible = tour;
      c.assis.x = c.assis.cible = assis ? 1 : 0;
      c.fondu.x = 1;
      for (int i = 0; i < 150; i++) {
        c.pas(1 / 120);
      }
      c.clin = 0;
      c.gx.x = 0;
      c.gy.x = 0;
      return c;
    }

    Widget mien(Corps c, {Offset dir = const Offset(0, -1)}) => SizedBox(
        width: 260,
        height: 268,
        child: CustomPaint(painter: _PosePeintre(c, Eclairage(couleur: Teinte.coeur, dir: dir, contre: 0.7))));
    Widget cellule(String nom, List<Widget> enfants) => Container(
          margin: const EdgeInsets.all(8),
          padding: const EdgeInsets.fromLTRB(12, 8, 12, 4),
          decoration: ShapeDecoration(color: Teinte.cielHorizon, shape: coin(20)),
          child: Column(mainAxisSize: MainAxisSize.min, children: <Widget>[
            Text(nom, style: texte(14, poids: FontWeight.w700)),
            Row(mainAxisSize: MainAxisSize.min, crossAxisAlignment: CrossAxisAlignment.end, children: enfants),
          ]),
        );
    final String quoi = Uri.base.queryParameters['planche'] ?? '1';
    // `?planche=avatar` : le portrait de Praxis, plein cadre, pour l'image de profil.
    // `fond=clair` : sur le lilas pâle au lieu du violet du logo.
    if (quoi == 'avatar') {
      return MaterialApp(
        debugShowCheckedModeBanner: false,
        home: SizedBox.expand(
          child: CustomPaint(painter: _AvatarPeintre(pose(Expr.sourire), Uri.base.queryParameters['fond'] == 'clair')),
        ),
      );
    }
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      theme: ThemeData(fontFamily: 'DMSans', scaffoldBackgroundColor: T.fond),
      home: Scaffold(
        body: SingleChildScrollView(
          child: Wrap(children: <Widget>[
            if (quoi == '1')
              for (final Expr e in Expr.values)
                cellule('${e.name} : origine à gauche, maquette à droite', <Widget>[
                  Padding(padding: const EdgeInsets.fromLTRB(30, 0, 30, 26), child: Praxis(doigt: doigt, taille: 200, depart: Humeur.calme, tombe: false)),
                  mien(pose(e)),
                ]),
            if (quoi == 'toque') ...<Widget>[
              for (final double t in <double>[0, 0.15, 0.3, 0.5, 0.7, 0.85, 1]) cellule('demi-tour : ${(t * 180).round()}°', <Widget>[mien(pose(Expr.sourire, tour: t))]),
              cellule('lumière à gauche', <Widget>[mien(pose(Expr.sourire), dir: const Offset(-0.7, -0.7))]),
              cellule('lumière à droite', <Widget>[mien(pose(Expr.sourire), dir: const Offset(0.7, -0.7))]),
              cellule('assis', <Widget>[mien(pose(Expr.attente, geste: Geste.genoux, assis: true))]),
              for (final Geste g in <Geste>[Geste.salut, Geste.pointe, Geste.montre, Geste.gene, Geste.un, Geste.cherche, Geste.reflexion, Geste.pause, Geste.balaie, Geste.coeur, Geste.desole, Geste.porte])
                cellule('geste : ${g.name}', <Widget>[mien(pose(Expr.sourire, geste: g))]),
            ],
          ]),
        ),
      ),
    );
  }
}

/// Le portrait : la tête de Praxis devant la lumière de la porte, sur le violet du logo.
class _AvatarPeintre extends CustomPainter {
  _AvatarPeintre(this.c, this.clair);
  final Corps c;
  final bool clair;
  @override
  void paint(Canvas canvas, Size s) {
    final double cote = math.min(s.width, s.height);
    final Rect cadre = Rect.fromLTWH(0, 0, cote, cote);
    canvas.clipRect(cadre);
    canvas.drawRect(cadre, Paint()..color = clair ? const Color(0xFFEEE6FA) : Teinte.violet);
    // La lumière de la porte, derrière lui : un seul dégradé.
    Atelier.etaler(canvas, Atelier.aura, Offset(cote * 0.5, cote * 0.50), cote * 0.74, cote * 0.74, clair ? 0.95 : 0.62);
    final double largeur = cote * 0.80;
    peindrePraxis(canvas, c, Offset(cote * 0.5, cote * 0.985), largeur, Eclairage(couleur: Teinte.coeur, dir: const Offset(0, -1), contre: 0.75));
  }

  @override
  bool shouldRepaint(_AvatarPeintre o) => false;
}

class _PosePeintre extends CustomPainter {

// ───────────────────────── Images par seconde ─────────────────────────
class Compteur extends StatefulWidget {
  const Compteur({super.key});
  @override
  State<Compteur> createState() => _CompteurState();
}

class _CompteurState extends State<Compteur> {
  final List<FrameTiming> _lot = <FrameTiming>[];
  String t = '…';
  DateTime _dernier = DateTime.now();
  @override
  void initState() {
    super.initState();
    SchedulerBinding.instance.addTimingsCallback(_recu);
  }

  void _recu(List<FrameTiming> f) {
    _lot.addAll(f);
    final DateTime m = DateTime.now();
    final int ms = m.difference(_dernier).inMilliseconds;
    if (ms < 800 || !mounted) return;
    final double moy = _lot.fold<int>(0, (int s, FrameTiming x) => s + x.totalSpan.inMicroseconds) / math.max(1, _lot.length) / 1000;
    final String n = '${(_lot.length * 1000 / ms).round()} images/s · ${moy.toStringAsFixed(1)} ms';
    _lot.clear();
    _dernier = m;
    setState(() => t = n);
  }

  @override
  void dispose() {
    SchedulerBinding.instance.removeTimingsCallback(_recu);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => IgnorePointer(
        child: Container(
          padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 5),
          decoration: ShapeDecoration(color: T.nuit.withValues(alpha: 0.78), shape: const StadiumBorder()),
          child: Text(t, style: texte(11.5, couleur: Colors.white, poids: FontWeight.w600).copyWith(fontFeatures: const <ui.FontFeature>[ui.FontFeature.tabularFigures()])),
        ),
      );
}
