  runApp(ValueListenableBuilder<bool>(

class Personne {
  const Personne(this.nom, this.pays, this.vu, this.piece, this.contrat, this.teinte);
  final String nom, pays, vu;
  final bool piece, contrat;
  final int teinte;
}

const List<Personne> gens = <Personne>[
  Personne('Amina B', 'Nigeria', 'il y a 2 h', true, true, 0), Personne('Youssouf D', 'Mali', 'il y a 5 h', false, true, 1),
  Personne('Khadija M', 'Maroc', 'hier', false, false, 2), Personne('Ibrahim S', 'Ghana', 'il y a 3 j', false, true, 3),
  Personne('Salma T', 'Égypte', 'il y a 20 min', true, true, 4), Personne('Moussa K', 'Guinée', 'il y a 6 j', false, false, 0),
  Personne('Hafsa A', 'Kenya', 'il y a 1 h', true, false, 1),
];

class Bureau extends StatefulWidget {
  const Bureau({super.key});
  @override
  State<Bureau> createState() => _BureauState();
}

class _BureauState extends State<Bureau> with SingleTickerProviderStateMixin<Bureau>, Vivant<Bureau> {
  final ValueNotifier<Offset> doigt = ValueNotifier<Offset>(Offset.zero);
  final GlobalKey<PraxisState> praxis = GlobalKey<PraxisState>();
  final GlobalKey<ParcoursState> fleuve = GlobalKey<ParcoursState>();
  final GlobalKey<IconeVivanteState> icMsg = GlobalKey<IconeVivanteState>(), icNotif = GlobalKey<IconeVivanteState>();
  final GlobalKey<_PileState> pile = GlobalKey<_PileState>();
  final ScrollController defile = ScrollController();
  final List<Bassin> bassins = <Bassin>[
    Bassin('À compléter', 206, ancien: 'le plus ancien : 9 j'),
    Bassin('En examen', 191, ancien: 'le plus ancien : 6 j'),
    Bassin('Entretien à planifier', 167, bloque: true),
    Bassin('Entretien choisi', -1),
    Bassin('Versement à débloquer', 33, remous: true),
  ];

  int? choisi;
  int page = 0, nbMsg = 3, nbNotif = 2, nbPieces = 17;
  String phrase = "167 personnes attendent un entretien, et aucun créneau n'est ouvert. C'est le seul blocage ce soir.";
  final List<(String, String, int)> messages = <(String, String, int)>[
    ('Youssouf D', "J'ai changé mon pays de versement, est-ce bon maintenant ?", 1),
    ('Salma T', 'Où puis-je signer le contrat ?', 4),
    ('Moussa K', "Aucun créneau d'entretien ne s'affiche.", 0),
  ];
  final R hauteur = R(196, k: 170, c: 15), volet = R(0, k: 200, c: 21), entete = R(0, k: 200, c: 24);
  double _baseHauteur = 196;
  Timer? _flux, _arrivee;
  @override
  Iterable<R> get ressorts => <R>[hauteur, volet, entete];

  @override
  void initState() {
    super.initState();
    defile.addListener(() {
      entete.cible = defile.offset > 12 ? 1 : 0;
      reveil();
    });
    _arrivee = Timer(const Duration(seconds: 7), () => recevoir(('Hafsa A', 'Bonjour, ma pièce est-elle bien arrivée ?', 1)));
  }

  @override
  void dispose() {
    _flux?.cancel();
    _arrivee?.cancel();
    defile.dispose();
    super.dispose();
  }

  void dire(String t, Humeur h, {Duration pour = const Duration(seconds: 3)}) {
    setState(() => phrase = t);
    praxis.currentState?.poser(h, pour: pour);
  }

  void avis(String t, {String? action, VoidCallback? surAction}) {
    late final OverlayEntry e;
    e = OverlayEntry(builder: (_) => Avis(t, action: action, surAction: surAction, surFin: () => e.remove()));
    Overlay.of(context).insert(e);
  }

  /// Un message tombe : le chiffre roule, l'icône remue, une bulle jaillit, un avis flotte.
  void recevoir((String, String, int) m) {
    if (!mounted) return;
    setState(() {
      messages.insert(0, m);
      nbMsg++;
    });
    icMsg.currentState?.remuer();
    praxis.currentState?.poser(Humeur.curieux, pour: const Duration(seconds: 2));
    avis('${m.$1} vous écrit');
  }

  void ouvrirCreneaux() {
    if (!bassins[2].bloque) return;
    setState(() {
      bassins[2].bloque = false;
      bassins[3].nb = 0;
    });
    dire("C'est ouvert. Les candidats sont prévenus et choisissent leur moment : regardez la barrière se lever.", Humeur.excite, pour: const Duration(seconds: 5));
    avis('12 créneaux ouverts cette semaine (exemple)');
    _flux = Timer.periodic(const Duration(milliseconds: 1300), (Timer t) {
      if (!mounted || bassins[3].nb >= 9) return t.cancel();
      setState(() {
        bassins[2].nb--;
        bassins[3].nb++;
      });
      fleuve.currentState?.secouer(3);
    });
  }

  void choisir(int? i) {
    setState(() => choisi = i);
    volet.cible = i == null ? 0 : 1;
    reveil();
    if (i != null) {
      dire(bassins[i].bloque ? "Ils n'attendent que vous : ouvrez des créneaux et ils choisissent seuls." : '${bassins[i].nom} : je vous montre les plus anciens en premier.',
          bassins[i].bloque ? Humeur.soucieux : Humeur.curieux);
    }
  }

  void ouvrirMessagerie() {
    final RenderBox b = icMsg.currentContext!.findRenderObject()! as RenderBox;
    final Offset centre = b.localToGlobal(b.size.center(Offset.zero));
    late final OverlayEntry e;
    e = OverlayEntry(builder: (_) => Eclosion(centre: centre, surFin: () => e.remove(), child: Messagerie(messages: messages)));
    Overlay.of(context).insert(e);
  }

  @override
  Widget build(BuildContext context) {
    Theme.of(context); // le bureau se redessine quand le thème change
    final Size ecran = MediaQuery.sizeOf(context);
    final bool large = ecran.width >= 900;
    final Widget contenu = SingleChildScrollView(
      controller: defile,
      padding: EdgeInsets.fromLTRB(large ? 28 : 16, 8, large ? 28 : 16, large ? 40 : 110),
      child: Center(
        child: ConstrainedBox(
          constraints: const BoxConstraints(maxWidth: 1240),
          child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: <Widget>[
            _ligne(large, <(int, Widget)>[
              (6, Accueil(praxis: praxis, doigt: doigt, phrase: phrase, actions: <Widget>[
                if (bassins[2].bloque) Pilule('Ouvrir des créneaux', fort: true, icone: Icons.event_available_rounded, surTouche: ouvrirCreneaux),
                Pilule('Vérifier les $nbPieces pièces', icone: Icons.badge_outlined, surTouche: () => pile.currentState?.appeler()),
                Pilule('Répondre aux $nbMsg messages', icone: Icons.chat_bubble_outline_rounded, surTouche: ouvrirMessagerie),
              ])),
              (5, Pile(
                key: pile,
                restant: nbPieces,
                surDecision: (Personne p, bool ok) {
                  setState(() => nbPieces--);
                  praxis.currentState?.poser(ok ? Humeur.heureux : Humeur.soucieux, pour: const Duration(milliseconds: 1600));
                  avis(ok ? 'Pièce de ${p.nom} validée' : 'Nouvelle pièce demandée à ${p.nom}', action: 'Annuler', surAction: () {
                    setState(() => nbPieces++);
                    pile.currentState?.rendre(p);
                  });
                },
              )),
            ]),
            const SizedBox(height: 22),
            Row(children: <Widget>[
              Text('Le parcours des candidats', style: titre(20)),
              const SizedBox(width: 10),
              Flexible(child: Text('Chaque tuile est large comme son groupe. Touchez-en une.', style: texte(13, couleur: T.doux), overflow: TextOverflow.ellipsis)),
            ]),
            const SizedBox(height: 12),
            GestureDetector(
              onScaleStart: (_) => _baseHauteur = hauteur.cible,
              onScaleUpdate: (ScaleUpdateDetails d) {
                if (d.pointerCount < 2 || !large) return;
                hauteur.cible = (_baseHauteur * d.verticalScale).clamp(170, 300);
                reveil();
              },
              child: Parcours(key: fleuve, bassins: bassins, choisi: choisi, hauteur: hauteur.x, surOuvrir: ouvrirCreneaux, surChoix: (int i) => choisir(choisi == i ? null : i)),
            ),
            if (large)
              MouseRegion(
                cursor: SystemMouseCursors.resizeRow,
                child: GestureDetector(
                  behavior: HitTestBehavior.opaque,
                  onVerticalDragUpdate: (DragUpdateDetails d) {
                    hauteur.cible = (hauteur.cible + d.delta.dy).clamp(170, 300);
                    hauteur.x = hauteur.cible;
                    reveil();
                  },
                  onVerticalDragEnd: (_) {
                    hauteur.v += 160; // un petit élan, puis le ressort le reprend
                    reveil();
                  },
                  child: SizedBox(height: 26, child: Center(child: Container(width: 54, height: 5, decoration: ShapeDecoration(color: T.filet, shape: const StadiumBorder())))),
                ),
              )
            else
              const SizedBox(height: 26),
            _ligne(large, <(int, Widget)>[
              (5, Bulles(messages: messages, surOuvrir: ouvrirMessagerie, surReponse: (int i) {
                setState(() {
                  messages.removeAt(i);
                  nbMsg = math.max(0, nbMsg - 1);
                });
                icMsg.currentState?.remuer();
                praxis.currentState?.poser(Humeur.heureux, pour: const Duration(milliseconds: 1400));
                avis('Réponse envoyée');
              })),
              (3, const Sante()),
              (5, const Activite()),
            ]),
            const SizedBox(height: 18),
            Text("Maquette : « À compléter », « En examen », « Entretien à planifier » et « Versement à débloquer » viennent de la production du 19/11. Les noms et les autres chiffres sont des exemples.",
                style: texte(12, couleur: T.doux)),
          ]),
        ),
      ),
    );
    return Listener(
      onPointerHover: (PointerHoverEvent e) => doigt.value = e.position,
      onPointerMove: (PointerMoveEvent e) => doigt.value = e.position,
      onPointerDown: (PointerDownEvent e) => doigt.value = e.position,
      child: Material(
        color: T.fond,
        child: Stack(children: <Widget>[
          Row(children: <Widget>[
            if (large) Rail(choix: page, surChoix: (int i) => setState(() => page = i)),
            Expanded(
              child: Column(children: <Widget>[
                Entete(
                  serre: entete.x,
                  large: large,
                  icMsg: icMsg,
                  icNotif: icNotif,
                  nbMsg: nbMsg,
                  nbNotif: nbNotif,
                  surMessages: ouvrirMessagerie,
                  surNotif: () {
                    setState(() => nbNotif++);
                    icNotif.currentState?.remuer();
                    avis('Amina B vient de déposer sa pièce');
                  },
                  surPersonne: (Personne p) => dire('${p.nom}, ${p.pays}. Dernière visite ${p.vu}.', Humeur.curieux),
                ),
                Expanded(child: contenu),
              ]),
            ),
          ]),
          if (!large) Positioned(left: 0, right: 0, bottom: 14, child: Center(child: BarreBas(choix: page, surChoix: (int i) => setState(() => page = i)))),
          if (volet.x > 0.004) ...<Widget>[
            Positioned.fill(
              child: GestureDetector(onTap: () => choisir(null), child: ColoredBox(color: T.nuit.withValues(alpha: 0.32 * volet.x.clamp(0, 1)))),
            ),
            if (large)
              Positioned(top: 12, bottom: 12, right: 12 - 440 * (1 - volet.x), width: 400, child: _volet())
            else
              Positioned(left: 0, right: 0, bottom: -ecran.height * 0.8 * (1 - volet.x), height: ecran.height * 0.78, child: _volet(bas: true)),
          ],
          Positioned(left: large ? 12 : null, right: large ? null : 12, bottom: large ? 12 : 78, child: const Compteur()),
        ]),
      ),
    );
  }

  Widget _volet({bool bas = false}) => Volet(
        bassin: bassins[choisi ?? 0],
        bas: bas,
        surFermer: () => choisir(null),
        surAccepter: (Personne p) {
          final int i = choisi ?? 0;
          setState(() => bassins[i].nb--);
          fleuve.currentState?.secouer(i);
          praxis.currentState?.poser(Humeur.heureux, pour: const Duration(milliseconds: 1500));
          avis('${p.nom} est acceptée', action: 'Annuler', surAction: () => setState(() => bassins[i].nb++));
        },
        surCreneaux: bassins[choisi ?? 0].bloque
            ? () {
                choisir(null);
                ouvrirCreneaux();
              }
            : null,
      );

  Widget _ligne(bool large, List<(int, Widget)> enfants) => large
      ? Row(crossAxisAlignment: CrossAxisAlignment.start, children: <Widget>[
          for (int i = 0; i < enfants.length; i++) ...<Widget>[if (i > 0) const SizedBox(width: 28), Expanded(flex: enfants[i].$1, child: enfants[i].$2)],
        ])
      : Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: <Widget>[
          for (int i = 0; i < enfants.length; i++) ...<Widget>[if (i > 0) const SizedBox(height: 26), enfants[i].$2],
        ]);
}

/// La phrase de Praxis s'écrit devant vous, puis ne bouge plus.
class Dit extends StatefulWidget {
  const Dit(this.phrase, {super.key});
  final String phrase;
  @override
  State<Dit> createState() => _DitState();
}

class _DitState extends State<Dit> with SingleTickerProviderStateMixin<Dit> {
  late final AnimationController a = AnimationController(vsync: this, duration: Duration(milliseconds: 14 * widget.phrase.length))..forward();
  @override
  void dispose() {
    a.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => AnimatedBuilder(
        animation: a,
        builder: (_, __) {
          final int n = (widget.phrase.length * a.value).round();
          return Text.rich(TextSpan(children: <InlineSpan>[
            TextSpan(text: widget.phrase.substring(0, n)),
            TextSpan(text: widget.phrase.substring(n), style: const TextStyle(color: Colors.transparent)),
          ]), style: texte(17, couleur: T.doux));
        },
      );
}

// ───────────────────────── La pile de pièces : on décide en glissant ─────────────────────────
class Pile extends StatefulWidget {
  const Pile({super.key, required this.restant, required this.surDecision});
  final int restant;
  final void Function(Personne, bool) surDecision;
  @override
  State<Pile> createState() => _PileState();
}

class _PileState extends State<Pile> with SingleTickerProviderStateMixin<Pile>, Vivant<Pile> {
  final List<Personne> cartes = <Personne>[gens[1], gens[2], gens[3], gens[5]];
  final R dx = R(0, k: 240, c: 17), saut = R(0, k: 260, c: 9);
  bool part = false;
  @override
  Iterable<R> get ressorts => <R>[dx, saut];
  void appeler() {
    saut.v += 5;
    reveil();
  }

  void rendre(Personne p) => setState(() {
        cartes.insert(0, p);
        saut.v += 4;
        reveil();
      });

  void decider(bool ok) {
    if (cartes.isEmpty || part) return;
    part = true;
    dx
      ..cible = ok ? 520 : -520
      ..k = 120;
    reveil();
    Future<void>.delayed(const Duration(milliseconds: 240), () {
      if (!mounted) return;
      final Personne p = cartes.removeAt(0);
      dx
        ..x = 0
        ..v = 0
        ..cible = 0
        ..k = 240;
      part = false;
      saut.v += 3.2;
      setState(() {});
      reveil();
      widget.surDecision(p, ok);
    });
  }

  @override
  Widget build(BuildContext context) {
    final double pencher = dx.x / 520;
    return Column(crossAxisAlignment: CrossAxisAlignment.start, children: <Widget>[
      Row(children: <Widget>[
        Rouleau(widget.restant, style: titre(34, couleur: T.accent)),
        const SizedBox(width: 10),
        Expanded(child: Text('pièces à vérifier\nGlissez à droite pour valider, à gauche pour redemander.', style: texte(13, couleur: T.doux))),
      ]),
      const SizedBox(height: 10),
      SizedBox(
        height: 212,
        child: cartes.isEmpty
            ? Center(child: Text('Plus aucune pièce. Rien ne vous attend ici.', style: texte(15, couleur: T.doux)))
            : Stack(alignment: Alignment.center, clipBehavior: Clip.none, children: <Widget>[
                for (int i = math.min(2, cartes.length - 1); i >= 0; i--)
                  if (i == 0)
                    Transform.translate(
                      offset: Offset(dx.x, -saut.x * 10 + pencher.abs() * 10),
                      child: Transform.rotate(
                        angle: pencher * 0.32,
                        child: GestureDetector(
                          onHorizontalDragUpdate: (DragUpdateDetails d) {
                            if (part) return;
                            dx.x += d.delta.dx;
                            dx.cible = dx.x;
                            reveil();
                          },
                          onHorizontalDragEnd: (DragEndDetails d) {
                            final double v = d.velocity.pixelsPerSecond.dx;
                            if (dx.x.abs() > 110 || v.abs() > 700) {
                              decider(dx.x + v * 0.1 > 0);
                            } else {
                              dx
                                ..cible = 0
                                ..v = v;
                              reveil();
                            }
                          },
                          child: Carte(cartes[0], tampon: (dx.x / 110).clamp(-1, 1)),
                        ),
                      ),
                    )
                  else
                    AnimatedSlide(
                      duration: const Duration(milliseconds: 520),
                      curve: Curves.elasticOut,
                      offset: Offset(0.04 * i, -0.055 * i),
                      child: AnimatedScale(
                        duration: const Duration(milliseconds: 520),
                        curve: Curves.elasticOut,
                        scale: 1 - 0.06 * i,
                        child: Transform.rotate(angle: 0.05 * i * (i.isOdd ? 1 : -1), child: Opacity(opacity: 1 - 0.25 * i, child: Carte(cartes[i]))),
                      ),
                    ),
              ]),
      ),
      const SizedBox(height: 4),
      Row(mainAxisAlignment: MainAxisAlignment.center, children: <Widget>[
        Pilule('Redemander', icone: Icons.west_rounded, couleur: T.ambrePale, encre: T.ambre, surTouche: () => decider(false)),
        const SizedBox(width: 10),
        Pilule('Valider', icone: Icons.east_rounded, couleur: T.vertPale, encre: T.vert, surTouche: () => decider(true)),
      ]),
    ]);
  }
}

                  child: Container(
                    padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 6),
                    decoration: ShapeDecoration(shape: coin(12).copyWithSide(tampon > 0 ? T.vert : T.ambre), color: T.surface.withValues(alpha: 0.9)),
                    child: Text(tampon > 0 ? 'VALIDÉE' : 'À REFAIRE', style: titre(18, couleur: tampon > 0 ? T.vert : T.ambre)),
                  ),
                ),
              ),
            ),
        ]),
      );
}

extension on ShapeBorder {
  ShapeBorder copyWithSide(Color c) =>

class _EauPeintre extends CustomPainter {
  _EauPeintre(this.n, this.h, this.t);
  final double n, h, t;
  @override
  void paint(Canvas c, Size s) {
    final Path rond = Path()..addOval((Offset.zero & s).deflate(3));
    c.drawPath(rond, Paint()..color = T.surface);
    c.save();
    c.clipPath(rond);
    final double y0 = s.height * (1 - n), a = 3.5 + h.abs() * 9;
    Path vague(double phase, double freq, double amp, double dy) {
      final Path p = Path()..moveTo(0, s.height);
      for (double x = 0; x <= s.width; x += 4) {
        final double k = x / s.width;
        p.lineTo(x, y0 + dy + math.sin(k * freq * 6.283 + phase) * amp + math.sin(k * freq * 13.2 - phase * 1.7) * amp * 0.35 + (k - 0.5) * h * 16);
      }
      return p..lineTo(s.width, s.height)..close();
    }

    c.drawPath(vague(t * 1.5, 1.1, a * 0.8, -3), Paint()..color = T.accent.withValues(alpha: 0.35));
    c.drawPath(vague(-t * 2.1, 1.4, a * 0.6, 2), Paint()..color = T.accent.withValues(alpha: 0.5));
    final Path devant = vague(t * 2.6, 1.0, a, 4);
    c.drawPath(devant, Paint()..shader = ui.Gradient.linear(Offset(0, y0), Offset(0, s.height), T.s ? const <Color>[Color(0xFFCDB5FC), Color(0xFF8E6BE0)] : const <Color>[Color(0xFF9C6CF2), Color(0xFF5B2FC0)]));
    final Paint bulle = Paint()..color = Colors.white.withValues(alpha: 0.35);
    for (int i = 0; i < 7; i++) {
      final double k = (t * (0.12 + i * 0.021) + i * 0.37) % 1, by = s.height - k * (s.height - y0 - 6);
      if (by > y0 + 8) c.drawCircle(Offset(s.width * (0.18 + i * 0.105) + math.sin(t * 2 + i) * 3, by), 1.6 + (i % 3), bulle);
    }
    c.drawPath(devant, Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2
      ..color = Colors.white.withValues(alpha: 0.45));
    c.restore();
    c.drawPath(rond, Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 3
      ..color = T.filet);
    final TextPainter tp = TextPainter(
        text: TextSpan(children: <InlineSpan>[
          TextSpan(text: '${(n * 900).round()}', style: titre(34, couleur: T.encre)),
          TextSpan(text: '\nsur 900', style: texte(13, couleur: T.doux, poids: FontWeight.w600)),
        ]),
        textAlign: TextAlign.center,
        textDirection: TextDirection.ltr)
      ..layout();
    tp.paint(c, Offset((s.width - tp.width) / 2, s.height * 0.16));
  }

  @override
  bool shouldRepaint(_EauPeintre o) => true;
}

class _LogoPeintre extends CustomPainter {
  _LogoPeintre(this.c1, this.c2);
  final Color c1, c2;
  @override
  void paint(Canvas c, Size s) {
    final double r = s.width / 2;
    c.drawCircle(Offset(r, r), r, Paint()..color = c1);
    final Path arche = Path()
      ..moveTo(r * 0.5, s.height)
      ..lineTo(r * 0.5, r * 1.05)
      ..arcToPoint(Offset(r * 1.5, r * 1.05), radius: Radius.circular(r * 0.5))
      ..lineTo(r * 1.5, s.height)
      ..close();
    c.save();
    c.clipPath(Path()..addOval(Offset.zero & s));
    c.drawPath(arche, Paint()..color = c2);
    c.drawRRect(RRect.fromRectAndCorners(Rect.fromLTWH(r * 0.68, r * 0.86, r * 0.64, r * 1.2), topLeft: Radius.circular(r * 0.32), topRight: Radius.circular(r * 0.32)), Paint()..color = c1);
    c.restore();
  }

  @override
  bool shouldRepaint(_LogoPeintre o) => o.c1 != c1 || o.c2 != c2;
}

// ───────────────────────── La page qui sort de l'icône, et y rentre ─────────────────────────
class Eclosion extends StatefulWidget {
  const Eclosion({super.key, required this.centre, required this.child, required this.surFin});
  final Offset centre;
  final Widget child;
  final VoidCallback surFin;
  @override
  State<Eclosion> createState() => EclosionState();
}

class EclosionState extends State<Eclosion> with SingleTickerProviderStateMixin<Eclosion>, Vivant<Eclosion> {
  final R r = R(0, k: 95, c: 17);
  bool ferme = false;
  @override
  Iterable<R> get ressorts => <R>[r];
  @override
  void initState() {
    super.initState();
    r.cible = 1;
    WidgetsBinding.instance.addPostFrameCallback((_) => reveil());
  }

  void fermer() {
    if (ferme) return;
    ferme = true;
    r
      ..cible = 0
      ..k = 150
      ..c = 24;
    reveil();
    Future<void>.delayed(const Duration(milliseconds: 520), widget.surFin);
  }

  @override
  Widget build(BuildContext context) {
    final Size s = MediaQuery.sizeOf(context);
    final double max = math.sqrt(s.width * s.width + s.height * s.height) + 40;
    return Positioned.fill(
      child: ClipPath(
        clipper: _Rond(widget.centre, math.max(0, r.x) * max),
        child: _Fermeture(fermer: fermer, child: widget.child),
      ),
    );
  }
}

class _Fermeture extends InheritedWidget {
  const _Fermeture({required this.fermer, required super.child});
  final VoidCallback fermer;
  @override
  bool updateShouldNotify(_Fermeture o) => false;
}

class _Rond extends CustomClipper<Path> {
  _Rond(this.c, this.r);
  final Offset c;
  final double r;
  @override
  Path getClip(Size s) => Path()..addOval(Rect.fromCircle(center: c, radius: r));
  @override
  bool shouldReclip(_Rond o) => o.r != r;
}

// ───────────────────────── Images par seconde ─────────────────────────
class Compteur extends StatefulWidget {
  const Compteur({super.key});
  @override
  State<Compteur> createState() => _CompteurState();
}

class _CompteurState extends State<Compteur> {
  final List<FrameTiming> _lot = <FrameTiming>[];
  String t = '…';
  DateTime _dernier = DateTime.now();
  @override
  void initState() {
    super.initState();
    SchedulerBinding.instance.addTimingsCallback(_recu);
  }

  void _recu(List<FrameTiming> f) {
    _lot.addAll(f);
    final DateTime m = DateTime.now();
    final int ms = m.difference(_dernier).inMilliseconds;
    if (ms < 800 || !mounted) return;
    final double moy = _lot.fold<int>(0, (int s, FrameTiming x) => s + x.totalSpan.inMicroseconds) / _lot.length / 1000;
    final String n = '${(_lot.length * 1000 / ms).round()} images/s · ${moy.toStringAsFixed(1)} ms';
    _lot.clear();
    _dernier = m;
    setState(() => t = n);
  }

  @override
  void dispose() {
    SchedulerBinding.instance.removeTimingsCallback(_recu);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => IgnorePointer(
        child: Container(
          padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),
          decoration: ShapeDecoration(color: T.nuit.withValues(alpha: 0.82), shape: const StadiumBorder()),
          child: Text(t, style: texte(12, couleur: Colors.white, poids: FontWeight.w600).copyWith(fontFeatures: const <ui.FontFeature>[ui.FontFeature.tabularFigures()])),
        ),
      );
}
