/// L'accueil famille sur le vrai routeur (comme accueil_famille_test.dart),
/// dans le cadre de capture.
Future<void> _monterFamille(
  WidgetTester tester,
  ThemeMode mode,
  double echelle,
) async {
  const String a1 = 'foyer-a-eleve-1';
  final FauxServiceAppairage appairage = FauxServiceAppairage()
    ..elevesRendus = FoyersFictifs.foyerA
    ..appareilsRendus = <AppareilAppaire>[
      AppareilAppaire(
        id: 'app-1',
        profilId: a1,
        prenom: 'Eleve A1',
        plateforme: 'Android',
        appaireA: DateTime.utc(2026, 9, 1),
      ),
    ];
  final FauxSeancesFamille seances = FauxSeancesFamille(
    seances: <SeanceFamille>[
      seanceFictive(
        seanceId: 's1',
        eleveId: a1,
        debutUtc: instantServeurFige.add(const Duration(hours: 3)),
        statut: StatutSeance.planifiee,
        heureFoyer: heureRendue('Europe/Paris', 'Paris', '18:10'),
      ),
      seanceFictive(
        seanceId: 's0',
        eleveId: a1,
        debutUtc: instantServeurFige.subtract(const Duration(days: 2)),
        statut: StatutSeance.terminee,
        heureFoyer: heureRendue(
          'Europe/Paris',
          'Paris',
          '18:00',
          date: 'Samedi 19 septembre',
        ),
      ),
    ],
  );
  late GoRouter routeur;
  await tester.pumpWidget(
    ProviderScope(
      overrides: <Override>[
        ...surchargesRouteur(
          FausseSourceSession(sessionActive(AppRole.guardian)),
          appairage: appairage,
        ),
        portSeancesFamilleProvider.overrideWithValue(seances),
        portClassesFoyerProvider.overrideWithValue(
          FauxClassesFoyer(
            classes: <ClasseFoyer>[
              ClasseFoyer(eleveId: a1, classe: classeFictive()),
            ],
          ),
        ),
        portTravailAVenirProvider.overrideWithValue(null),
      ],
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: Consumer(
            builder: (BuildContext _, WidgetRef ref, Widget? _) {
              routeur = ref.read(routeurProvider);
              return MaterialApp.router(
                debugShowCheckedModeBanner: false,
                theme: AppTheme.pour(InterfaceMode.standard, Brightness.light),
                darkTheme: AppTheme.pour(
                  InterfaceMode.standard,
                  Brightness.dark,
                ),
                themeMode: mode,
                builder: (BuildContext c, Widget? app) => MediaQuery(
                  data: MediaQuery.of(c)
                      .copyWith(textScaler: TextScaler.linear(echelle)),
                  child: app!,
                ),
                routerConfig: routeur,
              );
            },
          ),
        ),
      ),
    ),
  );
  await stabiliser(tester);
  routeur.go(Routes.tuteurAccueil);
  await stabiliser(tester);
}

void main() {
  group('Capture : BENTO-2c', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);

    String nom(double l, double e, ThemeMode m) =>
        '${l.toInt()}${e == 1 ? '' : 'x2'}_'
        '${m == ThemeMode.light ? 'clair' : 'sombre'}';

    for (final ThemeMode m in const <ThemeMode>[
      ThemeMode.light,
      ThemeMode.dark,
    ]) {
      for (final (double, double) c in const <(double, double)>[
        (320, 2),
        (390, 2),
        (1440, 1),
      ]) {
        testWidgets('journee 4 cahiers ${nom(c.$1, c.$2, m)}', (
          WidgetTester tester,
        ) async {
          fixerVueCapture(tester, Size(c.$1, c.$1 < 600 ? 3600 : 2600), 1);
          await tester.pumpWidget(
            hoteCapture(
              const JourneePage(),
              surcharges: _surchargesJournee(),
              themeMode: m,
              echelleTexte: c.$2,
            ),
          );
          await stabiliser(tester);
          expect(tester.takeException(), isNull);
          debugPrint(
            'PNG ${await capturer(tester, 'journee4_${nom(c.$1, c.$2, m)}')}',
          );
        });
      }
      for (final (double, double) c in const <(double, double)>[
        (320, 1),
        (320, 2),
        (600, 2),
        (620, 2),
        (863, 1),
        (864, 1),
        (1024, 1),
        (1440, 2),
      ]) {
        testWidgets('famille ${nom(c.$1, c.$2, m)}', (
          WidgetTester tester,
        ) async {
          fixerVueCapture(tester, Size(c.$1, 2600), 1);
          await _monterFamille(tester, m, c.$2);
          expect(tester.takeException(), isNull);
          debugPrint(
            'PNG ${await capturer(tester, 'famille_${nom(c.$1, c.$2, m)}')}',
          );
        });
      }
    }
  });
}
