/// Une adresse de demonstration sur ses vraies routes, dans le cadre de
/// capture.
Future<void> _monter(
  WidgetTester tester,
  String adresse,
  ThemeMode mode,
  double echelle,
) async {
  final GoRouter routeur = GoRouter(
    initialLocation: adresse,
    routes: routesDemoAdulte('dev'),
  );
  addTearDown(routeur.dispose);
  await tester.pumpWidget(
    ProviderScope(
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: MaterialApp.router(
            debugShowCheckedModeBanner: false,
            theme: AppTheme.pour(InterfaceMode.standard, Brightness.light),
            darkTheme: AppTheme.pour(InterfaceMode.standard, Brightness.dark),
            themeMode: mode,
            locale: AppLocale.fr.flutterLocale,
            supportedLocales: AppLocaleUtils.supportedLocales,
            localizationsDelegates: GlobalMaterialLocalizations.delegates,
            builder: (BuildContext c, Widget? app) => MediaQuery(
              data: MediaQuery.of(c)
                  .copyWith(textScaler: TextScaler.linear(echelle)),
              child: app!,
            ),
            routerConfig: routeur,
          ),
        ),
      ),
    ),
  );
  await stabiliser(tester);
}

void main() {
  group('Capture : BENTO-3c', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);

    for (final (double, double) cas in const <(double, double)>[
      (863, 1),
      (864, 1),
      (1440, 2),
    ]) {
      for (final ThemeMode m in const <ThemeMode>[
        ThemeMode.light,
        ThemeMode.dark,
      ]) {
        final String nom =
            'adulte_${cas.$1.toInt()}${cas.$2 == 1 ? '' : 'x2'}_'
            '${m == ThemeMode.light ? 'clair' : 'sombre'}';
        testWidgets(nom, (WidgetTester tester) async {
          fixerVueCapture(tester, Size(cas.$1, cas.$2 == 2 ? 2800 : 1500), 1);
          await _monter(
            tester,
            adresseDemoAdulte(EcranDemoAdulte.accueil),
            m,
            cas.$2,
          );
          expect(tester.takeException(), isNull);
          debugPrint('PNG ${await capturer(tester, nom)}');
        });
      }
    }
  });
}
