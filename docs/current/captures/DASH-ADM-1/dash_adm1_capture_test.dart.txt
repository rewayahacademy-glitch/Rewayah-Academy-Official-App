Widget _tableau() => TableauDeBordPage(horloge: () => DateTime(2026, 9, 8, 17));

List<Override> _surcharges({bool calme = false, bool erreurFile = false}) => <Override>[
  serviceAdminProvider.overrideWithValue(
    FauxServiceAdmin()
      ..indicateursRendus = calme ? const IndicateursAdmin(dossiersEnAttente: 0, foyersSansEnseignant: 0, rapportsManquants: 0, incidents24h: 2, comptesEchus: 1, demandesEnAttente: 0) : const IndicateursAdmin(
        dossiersEnAttente: 3,
        foyersSansEnseignant: 2,
        rapportsManquants: 1,
        incidents24h: 2,
        comptesEchus: 5,
        demandesEnAttente: 4,
      ),
  ),
  serviceValidationProvider.overrideWithValue(
    FauxServiceValidation()
      ..erreurFile = erreurFile ? const GenreLecteurInconnu() : null
      ..file = calme ? <DossierAValider>[] : <DossierAValider>[
        for (int i = 0; i < 3; i++)
          DossierAValider(
            documentId: 'doc$i',
            profilId: 'p$i',
            type: TypePiece.carteIdentite,
            cleObjet: 'p/$i.webp',
            deposeLe: DateTime(2026, 9, 5 + i),
            prenom: <String>['Abdourahmane', 'Mariama', 'Ibrahima'][i],
            nom: 'Barry',
            identifiantImmuable: 'TEACHER-MB-00000$i',
            role: AppRole.teacher,
          ),
      ],
  ),
];

void main() {
  group('Capture : DASH-ADM-1 Centre de commande', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);

    for (final (Size, ThemeMode, String, double, bool, bool) cas
        in <(Size, ThemeMode, String, double, bool, bool)>[
          (const Size(320, 1600), ThemeMode.light, 'adm1_320_clair', 1, false, false),
          (const Size(320, 3200), ThemeMode.light, 'adm1_320x2_clair', 2, false, false),
          (const Size(320, 3200), ThemeMode.dark, 'adm1_320x2_sombre', 2, false, false),
          (const Size(390, 1600), ThemeMode.dark, 'adm1_390_sombre', 1, false, false),
          (const Size(390, 1600), ThemeMode.light, 'adm1_390_calme_clair', 1, true, false),
          (const Size(390, 1600), ThemeMode.light, 'adm1_390_erreurliste_clair', 1, false, true),
          (const Size(1440, 1100), ThemeMode.light, 'adm1_1440_clair', 1, false, false),
          (const Size(1440, 1100), ThemeMode.dark, 'adm1_1440_sombre', 1, false, false),
          (const Size(1440, 2400), ThemeMode.light, 'adm1_1440x2_clair', 2, false, false),
          (const Size(1440, 2400), ThemeMode.dark, 'adm1_1440x2_sombre', 2, false, false),
        ]) {
      testWidgets(cas.$3, (WidgetTester tester) async {
        fixerVueCapture(tester, cas.$1, 1);
        await tester.pumpWidget(
          hoteCapture(
            _tableau(),
            surcharges: _surcharges(calme: cas.$5, erreurFile: cas.$6),
            themeMode: cas.$2,
            echelleTexte: cas.$4,
          ),
        );
        await stabiliser(tester);
        debugPrint('PNG ${await capturer(tester, cas.$3)}');
      });
    }
  });
}
