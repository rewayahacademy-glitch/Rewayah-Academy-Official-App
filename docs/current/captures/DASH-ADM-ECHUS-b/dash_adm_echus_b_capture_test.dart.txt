// retenue), puis apres un tap sur « Tous » (les echus relaches). LTR et RTL.
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/misc.dart' show Override;

class _Comptes implements ServiceComptes {
  @override
  Future<List<CompteAdmin>> comptes() async {
    final DateTime hier = DateTime.now().subtract(const Duration(days: 1));
    CompteAdmin c(
      String id,
      String prenom,
      AccountStatus etat,
      DateTime? fin,
    ) => CompteAdmin(
      profilId: id,
      identifiant: id,
      prenom: prenom,
      nom: 'Diallo',
      role: AppRole.guardian,
      etat: etat,
      inscritLe: DateTime(2026, 9, 1),
      accesJusquAu: fin,
      foyerId: null,
      piece: EtatPiece.validee,
      contrats: 1,
      aEnseignant: true,
      enfantsDuFoyer: 2,
    );
    return <CompteAdmin>[
      c('GUARDIAN-AD-000001', 'Aminata', AccountStatus.provisionalActive, hier),
      c('GUARDIAN-MD-000002', 'Mamadou', AccountStatus.provisionalActive, hier),
      c('GUARDIAN-FB-000003', 'Fatoumata', AccountStatus.active, null),
    ];
  }
}

void main() {
  group('Capture : DASH-ADM-ECHUS-b', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);
    for (final (Size, ThemeMode, double, TextDirection) cas
        in <(Size, ThemeMode, double, TextDirection)>[
          (const Size(390, 1200), ThemeMode.light, 1, TextDirection.ltr),
          (const Size(390, 1200), ThemeMode.light, 1, TextDirection.rtl),
          (const Size(320, 2400), ThemeMode.dark, 2, TextDirection.ltr),
          (const Size(1440, 900), ThemeMode.light, 1, TextDirection.ltr),
        ]) {
      final String suffixe =
          '${cas.$1.width.toInt()}${cas.$3 == 2 ? 'x2' : ''}_'
          '${cas.$2 == ThemeMode.light ? 'clair' : 'sombre'}'
          '${cas.$4 == TextDirection.rtl ? '_rtl' : ''}';
      testWidgets('comptes $suffixe', (WidgetTester tester) async {
        fixerVueCapture(tester, cas.$1, 1);
        await tester.pumpWidget(
          hoteCapture(
            Directionality(
              textDirection: cas.$4,
              child: const ComptesPage(echusSeulement: true),
            ),
            surcharges: <Override>[
              serviceComptesProvider.overrideWithValue(_Comptes()),
              serviceReseauProvider.overrideWithValue(FauxServiceReseau()),
            ],
            themeMode: cas.$2,
            echelleTexte: cas.$3,
          ),
        );
        await stabiliser(tester);
        debugPrint('PNG ${await capturer(tester, 'echus_b_filtre_$suffixe')}');
        await tester.tap(find.widgetWithText(FilterPill, 'Tous'));
        await stabiliser(tester);
        debugPrint('PNG ${await capturer(tester, 'echus_b_tous_$suffixe')}');
      });
    }
  });
}
