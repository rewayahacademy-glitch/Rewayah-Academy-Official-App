class _Comptes implements ServiceComptes {
  @override
  Future<List<CompteAdmin>> comptes() async {
    final DateTime hier = DateTime.now().subtract(const Duration(days: 1));
    CompteAdmin c(String id, String prenom, AccountStatus etat, DateTime? fin) =>
        CompteAdmin(
          profilId: id,
          identifiant: id,
          prenom: prenom,
          nom: 'Diallo',
          role: AppRole.guardian,
          etat: etat,
          inscritLe: DateTime(2026, 9, 1),
          accesJusquAu: fin,
          foyerId: null,
          piece: EtatPiece.validee,
          contrats: 1,
          aEnseignant: true,
          enfantsDuFoyer: 2,
        );
    return <CompteAdmin>[
      c('GUARDIAN-AD-000001', 'Aminata', AccountStatus.provisionalActive, hier),
      c('GUARDIAN-MD-000002', 'Mamadou', AccountStatus.provisionalActive, hier),
      c('GUARDIAN-FB-000003', 'Fatoumata', AccountStatus.active, null),
    ];
  }
}

void main() {
  group('Capture : DASH-ADM-ECHUS', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);
    for (final (Size, ThemeMode, double) cas in <(Size, ThemeMode, double)>[
      (const Size(390, 1600), ThemeMode.light, 1),
      (const Size(320, 3200), ThemeMode.dark, 2),
      (const Size(1440, 1100), ThemeMode.light, 1),
    ]) {
      final String suffixe =
          '${cas.$1.width.toInt()}${cas.$3 == 2 ? 'x2' : ''}_'
          '${cas.$2 == ThemeMode.light ? 'clair' : 'sombre'}';
      testWidgets('tableau $suffixe', (WidgetTester tester) async {
        fixerVueCapture(tester, cas.$1, 1);
        await tester.pumpWidget(
          hoteCapture(
            TableauDeBordPage(horloge: () => DateTime(2026, 9, 8, 17)),
            surcharges: <Override>[
              serviceAdminProvider.overrideWithValue(
                FauxServiceAdmin()
                  ..indicateursRendus = const IndicateursAdmin(
                    dossiersEnAttente: 0,
                    foyersSansEnseignant: 1,
                    rapportsManquants: 0,
                    incidents24h: 1,
                    comptesEchus: 2,
                    demandesEnAttente: 0,
                  ),
              ),
              serviceValidationProvider.overrideWithValue(
                FauxServiceValidation(),
              ),
            ],
            themeMode: cas.$2,
            echelleTexte: cas.$3,
          ),
        );
        await stabiliser(tester);
        debugPrint('PNG ${await capturer(tester, 'echus_tableau_$suffixe')}');
      });
      testWidgets('comptes $suffixe', (WidgetTester tester) async {
        fixerVueCapture(tester, cas.$1, 1);
        await tester.pumpWidget(
          hoteCapture(
            const ComptesPage(echusSeulement: true),
            surcharges: <Override>[
              serviceComptesProvider.overrideWithValue(_Comptes()),
              serviceReseauProvider.overrideWithValue(FauxServiceReseau()),
            ],
            themeMode: cas.$2,
            echelleTexte: cas.$3,
          ),
        );
        await stabiliser(tester);
        debugPrint('PNG ${await capturer(tester, 'echus_comptes_$suffixe')}');
      });
    }
  });
}
