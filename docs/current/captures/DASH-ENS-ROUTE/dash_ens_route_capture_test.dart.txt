CahierEnRetard _cahier(String id, String classe, int jour) => CahierEnRetard(
  seanceId: id,
  debut: DateTime(2026, 9, jour, 17),
  fin: DateTime(2026, 9, jour, 17, 45),
  nbParticipants: 3,
  heuresDeRetard: 30,
  groupeId: 'g-$id',
  groupeLibelle: classe,
);

List<Override> _surcharges(String etat) => <Override>[
  serviceReseauProvider.overrideWithValue(FauxServiceReseau()),
  // La file locale (drift) n'existe pas dans l'hote de capture.
  cahiersEnFileProvider.overrideWith((Ref _) async => const <String>{}),
  cahiersEnRetardProvider.overrideWith((Ref _) async {
    if (etat == 'erreur') throw StateError('reseau');
    if (etat == 'vide') return const <CahierEnRetard>[];
    return <CahierEnRetard>[
      _cahier('a', 'Al-Badr', 26),
      _cahier('b', 'As-Salam', 25),
      _cahier('c', 'Al-Fatiha', 23),
      _cahier('d', 'Al-Ikhlas', 22),
    ];
  }),
];

void main() {
  group('Capture : DASH-ENS-ROUTE', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);
    for (final (Size, ThemeMode, double, String) cas
        in <(Size, ThemeMode, double, String)>[
          (const Size(320, 1600), ThemeMode.light, 2, 'contenu'),
          (const Size(320, 1600), ThemeMode.dark, 2, 'contenu'),
          (const Size(390, 900), ThemeMode.light, 1, 'contenu'),
          (const Size(390, 900), ThemeMode.dark, 1, 'vide'),
          (const Size(390, 900), ThemeMode.light, 1, 'erreur'),
          (const Size(1440, 900), ThemeMode.light, 1, 'contenu'),
        ]) {
      final String nom =
          'a_terminer_${cas.$4}_${cas.$1.width.toInt()}'
          '${cas.$3 == 2 ? 'x2' : ''}_${cas.$2 == ThemeMode.light ? 'clair' : 'sombre'}';
      testWidgets(nom, (WidgetTester tester) async {
        fixerVueCapture(tester, cas.$1, 1);
        await tester.pumpWidget(
          hoteCapture(
            const ListeATerminerPage(),
            surcharges: _surcharges(cas.$4),
            themeMode: cas.$2,
            echelleTexte: cas.$3,
          ),
        );
        await stabiliser(tester);
        debugPrint('PNG ${await capturer(tester, nom)}');
      });
    }
  });
}
