Future<void> _monter(
  WidgetTester tester, {
  required ThemeMode theme,
  required double echelle,
  PortAccueilAccessibilite? port,
  bool horsLigne = false,
  String? aller,
}) async {
  late GoRouter routeur;
  await tester.pumpWidget(
    ProviderScope(
      overrides: <Override>[
        ...surchargesRouteur(
          FausseSourceSession(
            sessionActive(AppRole.student, mode: InterfaceMode.enfant),
          ),
        ),
        if (port != null)
          portAccueilAccessibiliteProvider.overrideWithValue(port),
        if (horsLigne)
          horsLigneProvider.overrideWith((Ref _) => Stream<bool>.value(true)),
      ],
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: Consumer(
            builder: (BuildContext _, WidgetRef ref, Widget? _) {
              routeur = ref.read(routeurProvider);
              return MaterialApp.router(
                debugShowCheckedModeBanner: false,
                theme: AppTheme.pour(InterfaceMode.enfant, Brightness.light),
                darkTheme: AppTheme.pour(InterfaceMode.enfant, Brightness.dark),
                themeMode: theme,
                routerConfig: routeur,
                locale: AppLocale.fr.flutterLocale,
                supportedLocales: AppLocaleUtils.supportedLocales,
                localizationsDelegates: GlobalMaterialLocalizations.delegates,
                builder: (BuildContext contexte, Widget? app) => MediaQuery(
                  data: MediaQuery.of(contexte)
                      .copyWith(textScaler: TextScaler.linear(echelle)),
                  child: app!,
                ),
              );
            },
          ),
        ),
      ),
    ),
  );
  await stabiliser(tester);
  if (aller != null) {
    routeur.go(aller);
    await stabiliser(tester);
  }
}

void main() {
  group('Capture : DASH-MA-PROD-b', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);

    const PortAccueilAccessibilite contenu = PortAccueilAccessibiliteDemo(
      eleve: EleveDemoAccessibilite.mariam,
    );
    final Map<String, (PortAccueilAccessibilite?, bool, String?)> etats =
        <String, (PortAccueilAccessibilite?, bool, String?)>{
          'vide': (
            const PortAccueilAccessibiliteDemo(
              eleve: EleveDemoAccessibilite.idriss,
              situation: SituationAccueilDemo.vide,
            ),
            false,
            null,
          ),
          'horsligne': (contenu, true, null),
        };
    final List<(String, Size, ThemeMode, double)> formes =
        <(String, Size, ThemeMode, double)>[
          ('320x2_clair', const Size(320, 3400), ThemeMode.light, 2),
          ('320x2_sombre', const Size(320, 3400), ThemeMode.dark, 2),
          ('390_clair', const Size(390, 1300), ThemeMode.light, 1),
          ('390_sombre', const Size(390, 1300), ThemeMode.dark, 1),
          ('1440_clair', const Size(1440, 1100), ThemeMode.light, 1),
        ];
    for (final MapEntry<String, (PortAccueilAccessibilite?, bool, String?)> e
        in etats.entries) {
      for (final (String nom, Size taille, ThemeMode theme, double echelle)
          in formes) {
        testWidgets('${e.key}_$nom', (WidgetTester tester) async {
          fixerVueCapture(tester, taille, 1);
          await _monter(
            tester,
            theme: theme,
            echelle: echelle,
            port: e.value.$1,
            horsLigne: e.value.$2,
            aller: e.value.$3,
          );
          debugPrint(
            'PNG ${await capturer(tester, 'ma_prod_b_${e.key}_$nom')}',
          );
        });
      }
    }
  });
}
