// ===== test/teaching_core/g/entree_explication_test.dart =====
  // EA-HL-DOUBLE-1d : l'echec de l'entree quand le reseau tombe apres la
  // verification (revue EA-HL-DOUBLE-1c-R, mineur 3), dans tous ses formats ;
  // et la verification hors ligne a 320 texte x2, LTR et RTL (mineur 2).
  // Sautees sans CAPTURE_DIR.
  group('Capture EA-HL-DOUBLE-1d', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);
    setUp(masquerRubanDebug);
    for (final (String, Size, ThemeMode, InterfaceMode, double, TextDirection)
        cas
        in <(String, Size, ThemeMode, InterfaceMode, double, TextDirection)>[
          (
            '390_clair',
            const Size(390, 844),
            ThemeMode.light,
            InterfaceMode.standard,
            1,
            TextDirection.ltr,
          ),
          (
            '390_sombre',
            const Size(390, 844),
            ThemeMode.dark,
            InterfaceMode.standard,
            1,
            TextDirection.ltr,
          ),
          (
            '390_guidee',
            const Size(390, 844),
            ThemeMode.light,
            InterfaceMode.enfant,
            1,
            TextDirection.ltr,
          ),
          (
            '320x2_clair',
            const Size(320, 1800),
            ThemeMode.light,
            InterfaceMode.standard,
            2,
            TextDirection.ltr,
          ),
          (
            '320x2_rtl_clair',
            const Size(320, 1800),
            ThemeMode.light,
            InterfaceMode.standard,
            2,
            TextDirection.rtl,
          ),
          (
            '1440_clair',
            const Size(1440, 900),
            ThemeMode.light,
            InterfaceMode.standard,
            1,
            TextDirection.ltr,
          ),
        ]) {
      testWidgets('echec de l\'entree ${cas.$1}', (WidgetTester tester) async {
        final FauxPortVisite visites = FauxPortVisite(
          echecEntree: const EchecObservation(MotifEchecObservation.reseau),
        );
        final FauxPortPolitique politique = FauxPortPolitique();
        addTearDown(politique.fermer);
        await _monter(
          tester,
          visites: visites,
          politique: politique,
          horsLigne: true,
          taille: cas.$2,
          mode: cas.$3,
          interface: cas.$4,
          echelle: cas.$5,
          direction: cas.$6,
        );
        await _presser(tester, find.text(t.classe.observer));
        await _presser(tester, _observerMaintenant);
        expect(tester.takeException(), isNull);
        expect(
          find.text(t.tc.observation.connexion.impossibleTitre),
          findsOneWidget,
        );
        expectHorsLigneUneFois();
        verifierAucunMotCoupe(tester, find.byType(MaterialApp));
        debugPrint(
          'PNG ${await capturerVue(tester, 'ea_hl_double1d_observation_echec_entree_${cas.$1}')}',
        );
      });

      if (cas.$5 == 2) {
        testWidgets('verification ${cas.$1}', (WidgetTester tester) async {
          final FauxPortPolitique politique = FauxPortPolitique()
            ..echecBudget = const EchecObservation(
              MotifEchecObservation.reseau,
            );
          addTearDown(politique.fermer);
          await _monter(
            tester,
            politique: politique,
            horsLigne: true,
            taille: cas.$2,
            mode: cas.$3,
            interface: cas.$4,
            echelle: cas.$5,
            direction: cas.$6,
          );
          await _presser(tester, find.text(t.classe.observer));
          expect(tester.takeException(), isNull);
          expectHorsLigneUneFois();
          verifierAucunMotCoupe(tester, find.byType(MaterialApp));
          debugPrint(
            'PNG ${await capturerVue(tester, 'ea_hl_double1d_observation_verification_${cas.$1}')}',
          );
        });
      }
    }
  });

    // sans et avec donnees (revue EA-HL-DOUBLE-1c-R, mineur 2). Sautees

      setUpAll(chargerPolicesDuDepot);
      setUp(masquerRubanDebug);
      for (final (String, TextDirection) cas in <(String, TextDirection)>[
        ('320x2_clair', TextDirection.ltr),
        ('320x2_rtl_clair', TextDirection.rtl),
      ]) {
        for (final bool sans in <bool>[true, false]) {
          final String etat = sans ? 'sans_donnees' : 'avec_donnees';
          testWidgets('$etat ${cas.$1}', (WidgetTester tester) async {
            await monterPage(
              tester,
              eleves: sans
                  ? () => Future<List<EleveDuFoyer>>.error(Exception('x'))
                  : () async => const <EleveDuFoyer>[],

            await tester.pumpAndSettle();
            expect(tester.takeException(), isNull);
            if (sans) {
              expectHorsLigneUneFois();
              expect(find.byType(EtatSansDonnees), findsNWidgets(2));
            } else {
              expectAvecDonnees();

              'PNG ${await capturerVue(tester, 'ea_hl_double1d_foyer_${etat}_${cas.$1}')}',
