  // EA-HL-DOUBLE-1e : captures de preuve de l'echec partiel de la Journee,
  // hors ligne (sous le bandeau) et en ligne, 390, 1440, sombre, 320 texte
  // x2 en LTR et RTL. Sautees sans CAPTURE_DIR.
  group('Capture EA-HL-DOUBLE-1e', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);
    setUp(masquerRubanDebug);
    for (final (String, Size, Brightness, double, TextDirection, bool) cas
        in <(String, Size, Brightness, double, TextDirection, bool)>[
          (
            '390_clair',
            const Size(390, 1400),
            Brightness.light,
            1,
            TextDirection.ltr,
            true,
          ),
          (
            '390_sombre',
            const Size(390, 1400),
            Brightness.dark,
            1,
            TextDirection.ltr,
            true,
          ),
          (
            '320x2_clair',
            const Size(320, 2600),
            Brightness.light,
            2,
            TextDirection.ltr,
            true,
          ),
          (
            '320x2_rtl_clair',
            const Size(320, 2600),
            Brightness.light,
            2,
            TextDirection.rtl,
            true,
          ),
          (
            '1440_clair',
            const Size(1440, 1000),
            Brightness.light,
            1,
            TextDirection.ltr,
            true,
          ),
          (
            '390_en_ligne_clair',
            const Size(390, 1400),
            Brightness.light,
            1,
            TextDirection.ltr,
            false,
          ),
        ]) {
      testWidgets(cas.$1, (WidgetTester tester) async {
        _fenetre(tester, cas.$2);
        await tester.pumpWidget(
          _hoteCarte(
            luminosite: cas.$3,
            echelleTexte: cas.$4,
            direction: cas.$5,
            surcharges: _surcharges(
              priere: _PriereEnPanne(),
              reseau: FauxServiceReseau(horsLigneRendu: cas.$6),
              enseignant: FauxServiceEnseignant()
                ..seancesRendues = <SeanceEnseignant>[_anNour()],
            ),
          ),
        );
        for (int i = 0; i < 10; i++) {
          await tester.pump(const Duration(milliseconds: 100));
        }
        expect(tester.takeException(), isNull);
        expect(find.text(t.enseignant.journee.erreurPartielle), findsOneWidget);
        if (cas.$6) {
          expectHorsLigneUneFois();
        }
        verifierAucunMotCoupe(tester, find.byType(MaterialApp));
        debugPrint(
          'PNG ${await capturerVue(tester, 'ea_hl_double1e_journee_partielle_${cas.$1}')}',
        );
      });
    }

    // Les cas « cahiers seuls » et « tout » (revue adverse du 1e), hors
    // ligne sous le bandeau, a 390 et a 320 texte x2 en LTR et RTL.
    for (final (String, bool) panne in <(String, bool)>[
      ('cahiers', false),
      ('tout', true),
    ]) {
      for (final (String, Size, double, TextDirection) cas
          in <(String, Size, double, TextDirection)>[
            ('390_clair', const Size(390, 1400), 1, TextDirection.ltr),
            ('320x2_clair', const Size(320, 2600), 2, TextDirection.ltr),
            ('320x2_rtl_clair', const Size(320, 2600), 2, TextDirection.rtl),
          ]) {
        testWidgets('${panne.$1} ${cas.$1}', (WidgetTester tester) async {
          _fenetre(tester, cas.$2);
          await tester.pumpWidget(
            _hoteCarte(
              luminosite: Brightness.light,
              echelleTexte: cas.$3,
              direction: cas.$4,
              surcharges: _surcharges(
                priere: panne.$2 ? _PriereEnPanne() : null,
                reseau: FauxServiceReseau(horsLigneRendu: true),
                enseignant: _CahiersEnPanne()
                  ..seancesRendues = <SeanceEnseignant>[_anNour()],
              ),
            ),
          );
          for (int i = 0; i < 10; i++) {
            await tester.pump(const Duration(milliseconds: 100));
          }
          expect(tester.takeException(), isNull);
          expect(
            find.text(
              panne.$2
                  ? t.enseignant.journee.erreurPartielleTout
                  : t.enseignant.journee.erreurPartielleCahiers,
            ),
            findsOneWidget,
          );
          expectHorsLigneUneFois();
          verifierAucunMotCoupe(tester, find.byType(MaterialApp));
          debugPrint(
            'PNG ${await capturerVue(tester, 'ea_hl_double1e_journee_partielle_${panne.$1}_${cas.$1}')}',
          );
        });
      }
    }
  });
}

// ===== test/features/accueil_adulte/accueil_adulte_page_test.dart =====
  // EA-HL-DOUBLE-1e (K-59, decide par delegation le 05/10, DIR-REP-1) :
  // l'echec de lecture de l'Accueil et de l'Historique dit la cause probable
  // sans l'affirmer ; « le serveur n'a pas repondu » est inexact quand c'est
  // l'appareil qui est hors ligne.
  group('Echec en ligne : cause probable non affirmee (EA-HL-DOUBLE-1e)', () {
    final List<(String, Widget, Override, String Function())> pages =
        <(String, Widget, Override, String Function())>[
          (
            'accueil',
            const AccueilAdultePage(),
            accueilAdulteProvider.overrideWith(
              (Ref _) => Future<AccueilAdulteVM>.error(StateError('lecture')),
            ),
            () => t.adulte.accueil.echec,
          ),
          (
            'historique',
            HistoriqueSeancesAdultePage(onRetour: () {}),
            historiqueSeancesAdulteProvider.overrideWith(
              (Ref _) =>
                  Future<List<SeancePasseeAdulte>>.error(StateError('lecture')),
            ),
            () => t.adulte.historique.echec,
          ),
        ];
    for (final (String, Widget, Override, String Function()) cas in pages) {
      testWidgets(cas.$1, (WidgetTester tester) async {
        await tester.pumpWidget(
          app(
            overrides: <Override>[
              horsLigneProvider.overrideWith(
                (Ref _) => Stream<bool>.value(false),
              ),
              cas.$3,
            ],
            child: cas.$2,
          ),
        );
        await tester.pumpAndSettle();
        final String texte = cas.$4();
        expect(find.text(texte), findsOneWidget);
        expect(texte, isNot(contains('serveur')));
        expect(texte, isNot(contains('pas répondu')));
        expect(
          texte,
          contains(
            "il est possible que la demande ne soit pas parvenue jusqu'à Rewayah",
          ),
        );
        expect(find.text(t.commun.reessayer), findsWidgets);
      });
    }

    group('Capture EA-HL-DOUBLE-1e', skip: sautSansCapture, () {
      setUpAll(chargerPolicesDuDepot);
      for (final (String, Widget, Override, String Function()) page in pages) {
        for (final (String, Size, double, TextDirection) cas
            in <(String, Size, double, TextDirection)>[
              ('390_clair', const Size(390, 844), 1, TextDirection.ltr),
              ('320x2_clair', const Size(320, 900), 2, TextDirection.ltr),
              ('320x2_rtl_clair', const Size(320, 900), 2, TextDirection.rtl),
            ]) {
          testWidgets('${page.$1} ${cas.$1}', (WidgetTester tester) async {
            fixerVueCapture(tester, cas.$2, 1);
            await tester.pumpWidget(
              hoteCapture(
                Directionality(textDirection: cas.$4, child: page.$2),
                surcharges: <Override>[
                  horsLigneProvider.overrideWith(
                    (Ref _) => Stream<bool>.value(false),
                  ),
                  page.$3,
                ],
                echelleTexte: cas.$3,
              ),
            );
            await stabiliser(tester);
            expect(tester.takeException(), isNull);
            expect(find.text(page.$4()), findsOneWidget);
            verifierAucunMotCoupe(tester, find.byType(MaterialApp));
            debugPrint(
              'PNG ${await capturer(tester, 'ea_hl_double1e_adulte_${page.$1}_echec_${cas.$1}')}',
            );
          });
        }
      }
    });
  });
