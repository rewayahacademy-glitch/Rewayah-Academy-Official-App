class _PortEnAttente implements PortAccueilAccessibilite {
  @override
  Future<DonneesAccueilAccessibilite> lire() =>
      Completer<DonneesAccueilAccessibilite>().future;
}

Future<void> _monter(
  WidgetTester tester, {
  required ThemeMode theme,
  required double echelle,
  required TextDirection sens,
  required PortAccueilAccessibilite port,
  bool horsLigne = false,
}) async {
  await tester.pumpWidget(
    ProviderScope(
      overrides: <Override>[
        ...surchargesRouteur(
          FausseSourceSession(
            sessionActive(AppRole.student, mode: InterfaceMode.enfant),
          ),
        ),
        portAccueilAccessibiliteProvider.overrideWithValue(port),
        if (horsLigne)
          horsLigneProvider.overrideWith((Ref _) => Stream<bool>.value(true)),
      ],
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: Consumer(
            builder: (BuildContext _, WidgetRef ref, Widget? _) =>
                MaterialApp.router(
                  debugShowCheckedModeBanner: false,
                  theme: AppTheme.pour(InterfaceMode.enfant, Brightness.light),
                  darkTheme: AppTheme.pour(
                    InterfaceMode.enfant,
                    Brightness.dark,
                  ),
                  themeMode: theme,
                  routerConfig: ref.read(routeurProvider),
                  locale: AppLocale.fr.flutterLocale,
                  supportedLocales: AppLocaleUtils.supportedLocales,
                  localizationsDelegates: GlobalMaterialLocalizations.delegates,
                  builder: (BuildContext contexte, Widget? app) => MediaQuery(
                    data: MediaQuery.of(contexte)
                        .copyWith(textScaler: TextScaler.linear(echelle)),
                    child: Directionality(textDirection: sens, child: app!),
                  ),
                ),
          ),
        ),
      ),
    ),
  );
  // Le squelette a son delai d'apparition (`Skeleton`) : il est passe.
  await stabiliser(tester);
}
