// Captures de MA-2COL-1b : l'Accueil du Mode Accessibilite sur la VRAIE
// route de l'eleve (`/eleve`), en echec de lecture, reessais epuises. Le
// seuil des deux colonnes est franchi SANS remonter l'application (fenetre
// redimensionnee, puis facteur de texte change) : le bloc garde son etat,
// la phrase « reessais epuises » reste et « Reessayer » ne revient pas.
// Puis le contenu, une colonne -> deux colonnes par redimensionnement.
//
// Saute sans CAPTURE_DIR : la CI ne l'execute jamais et n'ecrit rien.

import 'package:flutter/material.dart';
import 'package:flutter_localizations/flutter_localizations.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_riverpod/misc.dart' show Override;
import 'package:flutter_test/flutter_test.dart';
import 'package:rewayah_academy/core/accessibilite/port_accueil_accessibilite.dart';
import 'package:rewayah_academy/core/domain/enums.dart';
import 'package:rewayah_academy/core/i18n/strings.g.dart';
import 'package:rewayah_academy/core/router/app_router.dart';
import 'package:rewayah_academy/core/theme/app_theme.dart';
import 'package:rewayah_academy/features/demo/accessibilite_demo.dart';

import '../router/support_routeur.dart' show sessionActive, surchargesRouteur;
import '../support/harnais.dart' show FausseSourceSession;
import 'outil_capture.dart';

Future<void> _monter(
  WidgetTester tester, {
  required PortAccueilAccessibilite port,
  required ValueNotifier<double> facteur,
  required TextDirection sens,
  required ThemeMode theme,
}) async {
  await tester.pumpWidget(
    ProviderScope(
      overrides: <Override>[
        ...surchargesRouteur(
          FausseSourceSession(
            sessionActive(AppRole.student, mode: InterfaceMode.enfant),
          ),
        ),
        portAccueilAccessibiliteProvider.overrideWithValue(port),
      ],
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: Consumer(
            builder: (BuildContext _, WidgetRef ref, Widget? _) =>
                MaterialApp.router(
                  debugShowCheckedModeBanner: false,
                  theme: AppTheme.pour(InterfaceMode.enfant, Brightness.light),
                  darkTheme: AppTheme.pour(
                    InterfaceMode.enfant,
                    Brightness.dark,
                  ),
                  themeMode: theme,
                  routerConfig: ref.read(routeurProvider),
                  locale: AppLocale.fr.flutterLocale,
                  supportedLocales: AppLocaleUtils.supportedLocales,
                  localizationsDelegates: GlobalMaterialLocalizations.delegates,
                  builder: (BuildContext contexte, Widget? app) =>
                      ValueListenableBuilder<double>(
                        valueListenable: facteur,
                        builder: (BuildContext contexte, double f, Widget? _) =>
                            MediaQuery(
                              data: MediaQuery.of(contexte)
                                  .copyWith(textScaler: TextScaler.linear(f)),
                              child: Directionality(
                                textDirection: sens,
                                child: app!,
                              ),
                            ),
                      ),
                ),
          ),
        ),
      ),
    ),
  );
  await stabiliser(tester);
}

void main() {
  group('Capture : MA-2COL-1b', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);

    const PortAccueilAccessibilite enEchec = PortAccueilAccessibiliteDemo(
      eleve: EleveDemoAccessibilite.mariam,
      situation: SituationAccueilDemo.erreur,
    );
    const PortAccueilAccessibilite contenu = PortAccueilAccessibiliteDemo(
      eleve: EleveDemoAccessibilite.mariam,
    );

    Future<void> epuiser(WidgetTester tester) async {
      for (int i = 0; i < 3; i++) {
        await tester.tap(find.text(t.commun.etats.actions.reessayer));
        await stabiliser(tester);
      }
    }

    for (final (TextDirection sens, ThemeMode theme, String suffixe)
        in <(TextDirection, ThemeMode, String)>[
          (TextDirection.ltr, ThemeMode.light, 'ltr_clair'),
          (TextDirection.rtl, ThemeMode.dark, 'rtl_sombre'),
        ]) {
      testWidgets('erreur epuisee, fenetre 390 -> 1440, $suffixe', (
        WidgetTester tester,
      ) async {
        final ValueNotifier<double> facteur = ValueNotifier<double>(1);
        addTearDown(facteur.dispose);
        fixerVueCapture(tester, const Size(390, 1200), 1);
        await _monter(
          tester,
          port: enEchec,
          facteur: facteur,
          sens: sens,
          theme: theme,
        );
        await epuiser(tester);
        debugPrint(
          'PNG ${await capturer(tester, 'ma_2col_b_erreur_epuisee_1_390_$suffixe')}',
        );
        fixerVueCapture(tester, const Size(1440, 1100), 1);
        await stabiliser(tester);
        debugPrint(
          'PNG ${await capturer(tester, 'ma_2col_b_erreur_epuisee_2_1440_$suffixe')}',
        );
      });
    }

    testWidgets('erreur epuisee, facteur 1 -> 2 a 1440', (
      WidgetTester tester,
    ) async {
      final ValueNotifier<double> facteur = ValueNotifier<double>(1);
      addTearDown(facteur.dispose);
      fixerVueCapture(tester, const Size(1440, 1400), 1);
      await _monter(
        tester,
        port: enEchec,
        facteur: facteur,
        sens: TextDirection.ltr,
        theme: ThemeMode.light,
      );
      await epuiser(tester);
      facteur.value = 2;
      await stabiliser(tester);
      debugPrint(
        'PNG ${await capturer(tester, 'ma_2col_b_erreur_epuisee_3_1440x2_ltr_clair')}',
      );
    });

    testWidgets('contenu, fenetre 390 -> 1440', (WidgetTester tester) async {
      final ValueNotifier<double> facteur = ValueNotifier<double>(1);
      addTearDown(facteur.dispose);
      fixerVueCapture(tester, const Size(390, 1500), 1);
      await _monter(
        tester,
        port: contenu,
        facteur: facteur,
        sens: TextDirection.ltr,
        theme: ThemeMode.light,
      );
      debugPrint(
        'PNG ${await capturer(tester, 'ma_2col_b_contenu_1_390_ltr_clair')}',
      );
      fixerVueCapture(tester, const Size(1440, 1100), 1);
      await stabiliser(tester);
      debugPrint(
        'PNG ${await capturer(tester, 'ma_2col_b_contenu_2_1440_ltr_clair')}',
      );
    });
  });
}
