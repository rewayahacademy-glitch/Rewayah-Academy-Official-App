// PT-B-CORAN-ADULTE : captures de l'onglet Coran de l'eleve adulte, atteint
// en touchant l'onglet de la vraie barre (K-07).
// - `demo_coran_*` : `/demo/adulte`, onglet Coran -> la VRAIE page Coran
//   (`MushafPage`) sur le corpus de demonstration, recopie mot pour mot de la
//   migration 0013 (`service_coran_demo.dart`), avec la portee racine que
//   `main.dart` pose au demarrage sur `/demo/adulte` ; aucun serveur.
// - `demo_aller_a_*` : la feuille « Aller a » (390, feuille du bas) et le
//   dialogue (1440), saisie « 2:255 » ; `demo_page42_390` : la page ouverte.
// - `reel_coran_*` : route reelle (vrai routeur, session d'eleve, capacite
//   vraie), `/eleve/adulte` -> onglet Coran -> `/eleve/coran`. Le service
//   serveur est REMPLACE par le meme corpus de demonstration (aucun serveur
//   en test) : la capture prouve la navigation et le vrai rendu du vrai
//   texte, pas une lecture sur staging.
// 320 px texte x2, 390 clair et sombre, 1440.
// Saute sans CAPTURE_DIR (voir LISEZ-MOI.md). A copier dans test/capture/.

Finder _onglet(String libelle) => find
    .descendant(of: find.byType(NavShell), matching: find.text(libelle))
    .last;

Future<void> _monterDemo(
  WidgetTester tester,
  Size taille,
  double echelle,
  ThemeMode theme,
) async {
  tester.view.physicalSize = taille;
  tester.view.devicePixelRatio = 1;
  addTearDown(tester.view.reset);
  final GoRouter routeur = GoRouter(
    initialLocation: adresseDemoAdulte(EcranDemoAdulte.accueil),
    routes: routesDemoAdulte('dev'),
  );
  addTearDown(routeur.dispose);
  await tester.pumpWidget(
    ProviderScope(
      overrides: surchargesRacineDemoAdulte(
        adresse: Uri.parse('https://exemple.test/#/demo/adulte'),
        environnement: 'dev',
      ),
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: MaterialApp.router(
            debugShowCheckedModeBanner: false,
            theme: AppTheme.pour(InterfaceMode.standard, Brightness.light),
            darkTheme: AppTheme.pour(InterfaceMode.standard, Brightness.dark),
            themeMode: theme,
            locale: AppLocale.fr.flutterLocale,
            supportedLocales: AppLocaleUtils.supportedLocales,
            localizationsDelegates: GlobalMaterialLocalizations.delegates,
            builder: (BuildContext c, Widget? app) => MediaQuery(
              data: MediaQuery.of(c)
                  .copyWith(textScaler: TextScaler.linear(echelle)),
              child: app!,
            ),
            routerConfig: routeur,
          ),
        ),
      ),
    ),
  );
  await stabiliser(tester);
  await tester.tap(_onglet(t.adulte.nav.coran));
  await stabiliser(tester);
  expect(find.byType(CoranDemoAdulte), findsOneWidget);
  expect(find.byType(MushafPage), findsOneWidget);
  expect(find.textContaining(t.eleve.coran.source), findsOneWidget);
  expect(find.text(t.eleve.coran.indisponible), findsNothing);
}

void main() {
  group('Capture : PT-B-CORAN-ADULTE', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);

    // nom : (taille logique, echelle du texte, theme)
    final Map<String, (Size, double, ThemeMode)> formats =
        <String, (Size, double, ThemeMode)>{
          '320x2_clair': (const Size(320, 900), 2, ThemeMode.light),
          '390_clair': (const Size(390, 844), 1, ThemeMode.light),
          '390_sombre': (const Size(390, 844), 1, ThemeMode.dark),
          '1440_clair': (const Size(1440, 900), 1, ThemeMode.light),
        };

    for (final MapEntry<String, (Size, double, ThemeMode)> f
        in formats.entries) {
      final (Size taille, double echelle, ThemeMode theme) = f.value;

      testWidgets('reel_${f.key}', (WidgetTester tester) async {
        final GoRouter routeur = await monterAccueilAdulteReel(
          tester,
          taille: taille,
          echelle: echelle,
          theme: theme,
          surcharges: <Override>[
            ...surchargesAccueilAdulte(accueil: Valeur(accueilDemo())),
            ...surchargesCoranDemoAdulte(const ServiceCoranDemo()),
          ],
          depart: Routes.eleve,
        );
        await tester.tap(_onglet(t.nav.eleve.coran));
        await stabiliser(tester);
        expect(
          routeur.routerDelegate.currentConfiguration.uri.toString(),
          Routes.eleveCoran,
        );
        expect(find.byType(MushafPage), findsOneWidget);
        expect(find.textContaining(t.eleve.coran.source), findsOneWidget);
        await capturer(tester, 'reel_coran_${f.key}');
      });

      testWidgets('demo_${f.key}', (WidgetTester tester) async {
        await _monterDemo(tester, taille, echelle, theme);
        await capturer(tester, 'demo_coran_${f.key}');
      });
    }

    for (final (String nom, Size taille) in <(String, Size)>[
      ('390', const Size(390, 844)),
      ('1440', const Size(1440, 900)),
    ]) {
      testWidgets('demo_aller_a_$nom', (WidgetTester tester) async {
        await _monterDemo(tester, taille, 1, ThemeMode.light);
        await tester.tap(find.text(t.tc.coran.navigation.ouvrir));
        await stabiliser(tester);
        await tester.enterText(find.byType(TextField), '2:255');
        await stabiliser(tester);
        expect(
          find.text(t.tc.coran.navigation.pageNumero(page: 42)),
          findsOneWidget,
        );
        await capturer(tester, 'demo_aller_a_$nom');
        if (nom == '390') {
          await tester.tap(
            find.text(t.tc.coran.navigation.pageNumero(page: 42)),
          );
          await stabiliser(tester);
          expect(
            find.textContaining(t.eleve.coran.page(page: '42', total: '604')),
            findsWidgets,
          );
          await capturer(tester, 'demo_page42_390');
        }
      });
    }
  });
}
