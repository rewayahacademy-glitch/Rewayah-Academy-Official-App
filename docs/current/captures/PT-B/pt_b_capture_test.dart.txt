Finder _onglet(String libelle) => find
    .descendant(of: find.byType(NavShell), matching: find.text(libelle))
    .last;

Future<void> _monterDemo(
  WidgetTester tester,
  Size taille,
  double echelle,
  ThemeMode theme,
  TextDirection sens,
) async {
  tester.view.physicalSize = taille;
  tester.view.devicePixelRatio = 1;
  addTearDown(tester.view.reset);
  final GoRouter routeur = GoRouter(
    initialLocation: adresseDemoAdulte(EcranDemoAdulte.accueil),
    routes: routesDemoAdulte('dev'),
  );
  addTearDown(routeur.dispose);
  await tester.pumpWidget(
    ProviderScope(
      child: TranslationProvider(
        child: RepaintBoundary(
          key: cadreCapture,
          child: MaterialApp.router(
            debugShowCheckedModeBanner: false,
            theme: AppTheme.pour(InterfaceMode.standard, Brightness.light),
            darkTheme: AppTheme.pour(InterfaceMode.standard, Brightness.dark),
            themeMode: theme,
            locale: AppLocale.fr.flutterLocale,
            supportedLocales: AppLocaleUtils.supportedLocales,
            localizationsDelegates: GlobalMaterialLocalizations.delegates,
            builder: (BuildContext c, Widget? app) => MediaQuery(
              data: MediaQuery.of(c)
                  .copyWith(textScaler: TextScaler.linear(echelle)),
              child: Directionality(textDirection: sens, child: app!),
            ),
            routerConfig: routeur,
          ),
        ),
      ),
    ),
  );
  await stabiliser(tester);
  await tester.tap(_onglet(t.adulte.nav.coran));
  await stabiliser(tester);
  expect(find.byType(CoranDemoAdulte), findsOneWidget);
  expect(find.text(t.adulte.coranDemo.titre), findsOneWidget);
  expect(find.byType(MushafPage), findsNothing);
}

void main() {
  group('Capture : PT-B', skip: sautSansCapture, () {
    late CorpusReference reference;
    setUpAll(() async {
      await chargerPolicesDuDepot();
      reference = CorpusReference.charger();
    });

    // nom : (taille logique, echelle du texte, theme, sens)
    final Map<String, (Size, double, ThemeMode, TextDirection)> formats =
        <String, (Size, double, ThemeMode, TextDirection)>{
          '390_clair': (
            const Size(390, 844),
            1,
            ThemeMode.light,
            TextDirection.ltr,
          ),
          '320x2_clair': (
            const Size(320, 900),
            2,
            ThemeMode.light,
            TextDirection.ltr,
          ),
          '320x2_rtl': (
            const Size(320, 900),
            2,
            ThemeMode.light,
            TextDirection.rtl,
          ),
          '1440_sombre': (
            const Size(1440, 900),
            1,
            ThemeMode.dark,
            TextDirection.ltr,
          ),
          '390_rtl': (
            const Size(390, 844),
            1,
            ThemeMode.light,
            TextDirection.rtl,
          ),
        };

    for (final MapEntry<String, (Size, double, ThemeMode, TextDirection)> f
        in formats.entries) {
      final (Size taille, double echelle, ThemeMode theme, TextDirection sens) =
          f.value;

      testWidgets('demo_${f.key}', (WidgetTester tester) async {
        await _monterDemo(tester, taille, echelle, theme, sens);
        await capturer(tester, 'demo_coran_${f.key}');
      });

      testWidgets('reel_${f.key}', (WidgetTester tester) async {
        final GoRouter routeur = await monterAccueilAdulteReel(
          tester,
          taille: taille,
          echelle: echelle,
          theme: theme,
          direction: sens,
          surcharges: <Override>[
            ...surchargesAccueilAdulte(accueil: Valeur(accueilDemo())),
            portCorpusCoranProvider.overrideWithValue(
              FauxCorpusCoran(reference),
            ),
            portLayoutMushafProvider.overrideWithValue(
              const LayoutLignesIndisponible(),
            ),
            portNavigationCoranProvider.overrideWithValue(
              FauxNavigationCoran(),
            ),
          ],
          depart: Routes.eleve,
        );
        await tester.tap(_onglet(t.nav.eleve.coran));
        await stabiliser(tester);
        expect(
          routeur.routerDelegate.currentConfiguration.uri.toString(),
          Routes.eleveCoran,
        );
        expect(find.byType(MushafPage), findsOneWidget);
        await capturer(tester, 'reel_coran_${f.key}');
      });
    }
  });
}
