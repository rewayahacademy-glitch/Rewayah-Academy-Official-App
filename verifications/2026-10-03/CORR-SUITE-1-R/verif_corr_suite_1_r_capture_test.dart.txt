List<Override> _surchargesClasses() {

    ),

      (Ref _) async => DonneesEnseignantDemo.groupes(),

    seancesEnseignantProvider.overrideWith(

    ),
    cahiersEnRetardProvider.overrideWith(
      (Ref _) async =>

    ),
    detailsClassesProvider.overrideWithValue(<String, DetailClasse>{
      badr.id: const DetailClasse(
        niveau: 2,
        langue: 'Français',
        remplacement: true,
      ),
    }),
  ];
}

/// Releve : mots coupes au milieu, ellipses, sortie d'ecran a gauche/droite.
List<String> _releve(WidgetTester tester, double largeur) {
  final List<String> defauts = <String>[];
  for (final Element e in find.byType(RichText).evaluate()) {
    final RenderParagraph p = e.renderObject! as RenderParagraph;
    if (!p.hasSize) continue;
    final String texte = p.text.toPlainText();
    final Rect r = p.localToGlobal(Offset.zero) & p.size;
    if (r.left < -0.5 || r.right > largeur + 0.5) {
      defauts.add('HORS ECRAN « $texte » $r');
    }
    if (p.didExceedMaxLines) defauts.add('ELLIPSE/TRONQUE « $texte »');
    for (final RegExpMatch m in RegExp(r'[^\s]+').allMatches(texte)) {
      final List<TextBox> boites = p.getBoxesForSelection(
        TextSelection(baseOffset: m.start, extentOffset: m.end),
      );
      final Set<int> lignes = <int>{for (final TextBox b in boites) b.top.round()};
      if (lignes.length > 1) {
        defauts.add('MOT COUPE « ${m.group(0)} » dans « $texte »');
      }
    }
  }
  return defauts;
}

      for (final double echelle in <double>[1, 2]) {
        for (final TextDirection sens in TextDirection.values) {
          final String nom =
              'mes_classes_${largeur.round()}_x${echelle.round()}_${sens.name}';
          testWidgets(nom, (WidgetTester tester) async {
            fixerVueCapture(tester, Size(largeur, 1700), 1);
            await tester.pumpWidget(
              hoteCapture(
                Directionality(
                  textDirection: sens,
                  child: const MesClassesPage(),
                ),
                surcharges: _surchargesClasses(),
                echelleTexte: echelle,
              ),
            );
            await stabiliser(tester);
            final Object? ex = tester.takeException();
            final Finder mot = find.text(t.enseignant.classes.cahierAValider);
            final Rect rMot = tester.getRect(mot);
            final Rect rPill = tester.getRect(
              find.text(t.enseignant.classes.remplacement),
            );
            debugPrint('RELEVE $nom exception=$ex mot=$rMot pastille=$rPill');
            for (final String d in _releve(tester, largeur)) {
              debugPrint('RELEVE $nom $d');
            }
            debugPrint('PNG ${await capturer(tester, nom)}');
          });
        }
      }
    }
  });
