Future<void> _monter(
  WidgetTester tester,
  Widget ecran, {
  ThemeMode mode = ThemeMode.light,
  double echelle = 2,
}) async {
  fixerVueCapture(tester, const Size(390, 844), 1);
  await tester.pumpWidget(
    hoteCapture(
      ecran,
      themeMode: mode,
      echelleTexte: echelle,
      interfaceMode: InterfaceMode.enfant,
    ),
  );
  await stabiliser(tester);
}

void main() {
  group('INTEG-RC-1-R : salle MA 390 x2', skip: sautSansCapture, () {
    setUpAll(chargerPolicesDuDepot);
    for (final EleveDemoAccessibilite eleve in EleveDemoAccessibilite.values) {
      for (final ThemeMode mode in <ThemeMode>[
        ThemeMode.light,
        ThemeMode.dark,
      ]) {
        testWidgets('salle_${eleve.name}_${mode.name}', (
          WidgetTester tester,
        ) async {
          await _monter(
            tester,
            EcranAccessibiliteDemo(
              eleve: eleve,
              moment: MomentDemoAccessibilite.fenetre,
              etapeInitiale: EtapeDemoAccessibilite.salle,
            ),
            mode: mode,
          );
          debugPrint(
            'PNG ${await capturer(tester, 'salle_ma_${eleve.name}_390_x2_${mode.name}')}',
          );
          debugPrint('EXCEPTION salle ${eleve.name}: ${tester.takeException()}');
        });
      }
    }
  });

  group('INTEG-RC-1-R : lecteur du Coran MA 390 x2', skip: sautSansCapture, () {
    late CorpusReference reference;
    setUpAll(() async {
      await chargerPolicesDuDepot();
      reference = CorpusReference.charger();
    });
    List<Override> corpus(int page) => <Override>[
      portCorpusCoranProvider.overrideWithValue(FauxCorpusCoran(reference)),
      portLayoutMushafProvider.overrideWithValue(
        const LayoutLignesIndisponible(),
      ),
      portNavigationCoranProvider.overrideWithValue(
        FauxNavigationCoran(
          personnelle: PositionLecture(page: page, sourate: 1, verset: 1),
        ),
      ),
    ];
    for (final (String nom, bool grande, bool premiere) in <(String, bool, bool)>[
      ('mushaf_derniere', false, false),
      ('verset_agrandi_derniere', true, false),
      ('mushaf_page1', false, true),
    ]) {
      testWidgets('coran_$nom', (WidgetTester tester) async {
        await _monter(
          tester,
          EcranAccessibiliteDemo(
            eleve: EleveDemoAccessibilite.mariam,
            destinationInitiale: EcranAccessibiliteDemo.indexCoran,
            surchargesCoran: corpus(
              premiere ? 1 : reference.pages.last.numero,
            ),
          ),
        );
        debugPrint('PNG ${await capturer(tester, 'coran_ma_entree_${nom}_390_x2')}');
        if (grande) {
          ProviderScope.containerOf(
            tester.element(find.byType(EcranAccessibiliteDemo)),
          ).read(preferencesLocalesProvider.notifier).lectureGrande(true);
          await tester.pump();
        }
        await tester.tap(
          find.byKey(const ValueKey<String>('coran-ma-principale')),
        );
        for (int i = 0; i < 10; i++) {
          await tester.pump(const Duration(milliseconds: 50));
        }
        debugPrint('PNG ${await capturer(tester, 'coran_ma_lecteur_${nom}_390_x2')}');
        debugPrint(
          'TOOLTIP precedente=${find.byTooltip('Page précédente').evaluate().length} '
          'suivante=${find.byTooltip('Page suivante').evaluate().length}',
        );
        await tester.pump(CommandesMasquables.delaiParDefaut);
        await tester.pump(const Duration(seconds: 1));
        final double coran = tester
            .getSize(find.byType(CommandesMasquables))
            .height;
        debugPrint(
          'CORAN $nom masque : ${coran.toStringAsFixed(0)} / 844 = '
          '${(coran / 844 * 100).toStringAsFixed(1)} %',
        );
        debugPrint(
          'PNG ${await capturer(tester, 'coran_ma_lecteur_${nom}_390_x2_masque')}',
        );
        debugPrint('EXCEPTION coran $nom: ${tester.takeException()}');
      });
    }
  });
}
